6BatteryTier Cloud::calculateBatteryTier(float currentSoC) {
7 // Get previous tier for hysteresis logic
8 uint8_t prevTierValue = sysStatus.get_currentBatteryTier();
9 BatteryTier prevTier = static_cast<BatteryTier>(prevTierValue);
10
11 // Apply hysteresis: require 5% higher SoC to move to better tier
12 // This prevents rapid tier thrashing when battery is near threshold
13
14 if (currentSoC >= 75.0f) {
15 // Strong signal to move to HEALTHY (or stay there)
16 return TIER_HEALTHY;
17 } else if (currentSoC >= 70.0f) {
18 // Boundary zone: stay in current tier unless coming from below
19 return (prevTier == TIER_HEALTHY) ? TIER_HEALTHY : TIER_CONSERVING;
20 } else if (currentSoC >= 55.0f) {
21 // Strong signal for CONSERVING or recovery from CRITICAL
22 return TIER_CONSERVING;
23 } else if (currentSoC >= 50.0f) {
24 // Boundary zone: stay in current tier unless coming from above
25 if (prevTier <= TIER_CONSERVING) {
26 return TIER_CONSERVING;
27 } else {
28 return TIER_CRITICAL;
29 }
30 } else if (currentSoC >= 35.0f) {
31 // Strong signal for CRITICAL or recovery from SURVIVAL
32 return TIER_CRITICAL;
33 } else if (currentSoC >= 30.0f) {
34 // Boundary zone: stay in current tier unless coming from above
35 if (prevTier <= TIER_CRITICAL) {
36 return TIER_CRITICAL;
37 } else {
38 return TIER_SURVIVAL;
39 }
40 } else {
41 // Below 30% - definite SURVIVAL mode
42 return TIER_SURVIVAL;
43 }
44}

47uint16_t Cloud::getIntervalMultiplier(BatteryTier tier) {
48 switch (tier) {
49 case TIER_HEALTHY:
50 return 1; // Normal interval
51 case TIER_CONSERVING:
52 return 2; // Double interval (half connections per day)
53 case TIER_CRITICAL:
54 return 4; // Quadruple interval (quarter connections per day)
55 case TIER_SURVIVAL:
56 return 12; // 12x interval (e.g., hourly → every 12 hours)
57 default:
58 Log.warn("Unknown battery tier %d, defaulting to 1x multiplier", (int)tier);
59 return 1;
60 }

64float Cloud::getConnectionBackoffMultiplier(uint16_t lastDurationSec) {
65 // Evaluate connection history to apply additional backoff
66 // in problematic locations (poor signal, tower congestion, etc.)
67
68 if (lastDurationSec == 0) {
69 // Failed connection or first attempt - apply strong backoff
70 Log.info("Connection history: failed/first attempt - applying 2.0x backoff");
71 return 2.0f;
72 } else if (lastDurationSec < 60) {
73 // Fast connection - excellent location
74 Log.trace("Connection history: fast (%ds) - no additional backoff", lastDurationSec);
75 return 1.0f;
76 } else if (lastDurationSec < 180) {
77 // Normal connection time - acceptable
78 Log.trace("Connection history: normal (%ds) - no additional backoff", lastDurationSec);
79 return 1.0f;
80 } else if (lastDurationSec < 300) {
81 // Slow connection - marginal location
82 Log.info("Connection history: slow (%ds) - applying 1.5x backoff", lastDurationSec);
83 return 1.5f;
84 } else {
85 // Very slow connection - problematic location
86 Log.info("Connection history: problem (%ds) - applying 2.0x backoff", lastDurationSec);
87 return 2.0f;
88 }
