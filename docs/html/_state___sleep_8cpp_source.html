891 }

893 // For COUNTING mode: Turn off LED if flash completed during sleep
894 if (sysStatus.get_sensorMode() == COUNTING && signalLEDTimeRemaining() == 0 && signalLEDStatus()) {
895 signalLED(false);
896 }
897
898 // Timer wake = scheduled report. No checks, no gates.
899 // We trust that the system timer woke us at the correct boundary.
900 if (timerWake) {

902 // while occupied so occupancy=1 is only reported on 0->1 transition.
903 if (sysStatus.get_sensorMode() == OCCUPANCY && current.get_occupied() &&

906 state = SLEEPING_STATE;
907 return;
908 }

910 Log.info("WAKE: Timer wake - reason=SCHEDULED_REPORT transitioning to REPORTING_STATE");
911 state = REPORTING_STATE;
912 return;
913 }

915 // For PIR wakes, check if reporting is also due (opportunistic reporting)
916 if (pirWake && Time.isValid() && isWithinOpenHours()) {
917 // In OCCUPANCY + INTERMITTENT_KEEP_ALIVE mode, do not opportunistically
918 // report while occupied; PIR hits should only reset debounce.
919 if (sysStatus.get_sensorMode() == OCCUPANCY && current.get_occupied() &&
920 sysStatus.get_connectionMode() == INTERMITTENT_KEEP_ALIVE) {
921 // Skip overdue-report check while occupied.
922 } else {
923 uint16_t intervalSec = sysStatus.get_reportingInterval();
924 if (intervalSec == 0) intervalSec = 3600;
925
926 time_t now = Time.now();
927 time_t lastReport = sysStatus.get_lastReport();
928 if (lastReport > 0 && (now - lastReport) >= intervalSec) {
929 int overdue = (int)(now - lastReport - intervalSec);
930 Log.info("WAKE: PIR + report overdue (%d sec) - transitioning to REPORTING_STATE", overdue);
931 state = REPORTING_STATE;
932 return;
933 }
934 }
935 }
936
937 // If PIR woke us in INTERMITTENT or DISCONNECTED mode and no report is needed,
938 // return immediately to sleep. This check comes AFTER opportunistic reporting
939 // so overdue reports are not missed.
940 if (pirWake && sysStatus.get_connectionMode() != CONNECTED) {
941 state = SLEEPING_STATE;
942 return;
943 }
944
945 Log.info("WAKE: No immediate action needed - transitioning to IDLE_STATE");
946 state = IDLE_STATE;
947 }
