138unsigned long computeCloudSyncTimeoutMs(uint16_t queueDepth) {
139 const unsigned long baseTimeoutMs = ConnectivityPolicy::CLOUD_OPS_GATE_TIMEOUT_MS;
140 if (queueDepth == 0) {

146 const unsigned long queueDrainCushionMs = 5000UL;
147 const unsigned long maxTimeoutMs = 120000UL;
148

150 publishStartupMs + ((unsigned long)queueDepth * perEventDrainMs) + queueDrainCushionMs;
151 if (queueAwareTimeoutMs < baseTimeoutMs) {
152 queueAwareTimeoutMs = baseTimeoutMs;
153 }
154 if (queueAwareTimeoutMs > maxTimeoutMs) {
155 queueAwareTimeoutMs = maxTimeoutMs;

157 return queueAwareTimeoutMs;

233bool clearAb1805StaleAlarmInterrupts() {
234 static const uint8_t kRegStatus = 0x0f;
235 static const uint8_t kRegIntMask = 0x12;
236 static const uint8_t kMaskAieTie = 0x0c;
237
238 uint8_t intMask = 0;
239 if (!ab1805.readRegister(kRegIntMask, intMask)) {
240 return false;
241 }
242
243 // Clear stale interrupt flags before arming the next one-shot alarm.
244 if (!ab1805.writeRegister(kRegStatus, 0x00)) {
245 return false;
246 }
247
248 intMask = (uint8_t)(intMask & (uint8_t)~kMaskAieTie);
249 if (!ab1805.writeRegister(kRegIntMask, intMask)) {
250 return false;
251 }
252
253 if (!ab1805.maskRegister(
254 AB1805::REG_TIMER_CTRL,
255 (uint8_t)~AB1805::REG_TIMER_CTRL_RPT_MASK,
256 AB1805::REG_TIMER_CTRL_RPT_DIS)) {
257 return false;
258 }
259
260 return true;
261}

349 // Determine if we will use *effective* network standby for the upcoming sleep.
350 // Only cellular platforms actually apply standby; on WiFi, treat as disabled
351 // so disconnect logic doesn't wait for a radio state that won't change.

353 // wake cycle; if connect never succeeded, preserving the modem state buys
354 // nothing and can carry a bad NCP state into the next wake.

359#if HAL_PLATFORM_CELLULAR

394 // ********** Cloud Operations Sync Prerequisites (with timeout budget) **********
395 // Before disconnecting, ensure critical operations complete:
396 // 1) Publish queue drained
397 // 2) Ledgers synced from cloud
398 // 3) OTA updates checked
399 // 4) Webhook response received

401 // NOTE: This code is non-blocking at application level (Particle.process() called)
402 // but blocks state transition until prerequisites complete or timeout.
403 static unsigned long cloudSyncStartMs = 0;
404 static unsigned long cloudSyncBudgetMs = 0;

406 static uint16_t lastQueueDepth = 0xFFFF;

409 // Device OS already manages the asynchronous cloud session teardown once
410 // Particle.disconnect() has been requested. Here, we keep application
411 // logic minimal: request cloud disconnect and radio-off once, then wait
412 // (bounded) until both cloud and modem are actually off before sleeping.
413 static bool disconnectRequested = false;
414 static unsigned long disconnectRequestStartMs = 0;
415 static unsigned long cloudDisconnectStartMs = 0;
416 static unsigned long cloudDisconnectElapsedMs = 0;
417 static unsigned long modemOffElapsedMs = 0;

419 static unsigned long lastGateProgressMs = 0;
