247uint32_t Cloud::noteLedgerSyncRequest(LedgerRequestKind kind,
248 const char *source,
249 const void *ptr) {
250 const unsigned long issueMs = millis();
251 LedgerRequestRecord *existing = findRequestByPointer(ptr);
252
253 bool inflightForPointer = false;
254 if (ptr == &deviceStatusLedger) {
255 inflightForPointer = ledgerHasUnsyncedWriteForDiag(deviceStatusLedger);
256 } else if (ptr == &deviceDataLedger) {
257 inflightForPointer = ledgerHasUnsyncedWriteForDiag(deviceDataLedger);
258 }
259
260 if (existing && inflightForPointer) {
261 const unsigned long ageMs = issueMs - existing->issueMs;
262#if ENABLE_LEDGER_TRACE
263 Log.warn("LedgerDuplicateIssue: source=%s ptr=%p existingSeq=%lu age=%lu",
264 ledgerRequestSourceLabel(source),
265 ptr,
266 (unsigned long)existing->seq,
267 ageMs);
268 Log.info("LedgerIssueSkip: source=%s ptr=%p existingSeq=%lu age=%lu",
269 ledgerRequestSourceLabel(source),
270 ptr,
271 (unsigned long)existing->seq,
272 ageMs);
273#else
274 if (ageMs > 30000UL) {
275 Log.warn("LedgerDuplicateStillInflight: source=%s ptr=%p existingSeq=%lu age=%lu",
276 ledgerRequestSourceLabel(source),
277 ptr,
278 (unsigned long)existing->seq,
279 ageMs);
280 }
281#endif
282 return 0;
283 }
284
285 const uint32_t seq = ++ledgerRequestSequence;
286
287 if (ledgerRequestCount < LEDGER_REQUEST_TRACKER_MAX) {
288 ledgerRequests[ledgerRequestCount++] = {seq, kind, source, ptr, issueMs};
289 } else {
290 Log.warn("LedgerIssueTrackerFull: seq=%lu kind=%s active=%u", seq, ledgerRequestKindLabel(kind), (unsigned)ledgerRequestCount);
291 }
292
293 if (!ledgerDrainActive) {
294 ledgerDrainActive = true;
295 ledgerDrainStartMs = issueMs;
296 ledgerCompletedAgeMinMs = 0;
297 ledgerCompletedAgeMaxMs = 0;
298 ledgerCompletedAgeValid = false;
299 }
300
301 const uint16_t pendingCount = currentLedgerPendingCount();
302 const bool activeNow = pendingCount > 0;
303#if ENABLE_LEDGER_TRACE
304 Log.info("LedgerIssue: seq=%lu kind=%s source=%s ptr=%p pending=%u syncing=%d inflight=%d ms=%lu",
305 (unsigned long)seq,
306 ledgerRequestKindLabel(kind),
307 ledgerRequestSourceLabel(source),
308 ptr,
309 (unsigned)pendingCount,
310 activeNow ? 1 : 0,
311 activeNow ? 1 : 0,
312 issueMs);
313#else
314 (void)pendingCount;
315 (void)activeNow;
316#endif
317
318 return seq;
319}

575void Cloud::loop() {
576 // Apply any newly-synced configuration outside callback context.
577 // Do at most one deferred operation per loop() pass.
578 if (pendingConfigApply && Particle.connected()) {
579 pendingConfigApply = false;
580 LedgerData defaults = defaultSettingsLedger.get();
581 LedgerData device = deviceSettingsLedger.get();
582 lastApplySuccess = applyConfigurationFromLedger(defaults, device);
583
584 if (lastApplySuccess) {
585 lastApplySuccess = finalizeLedgerAppliedConfig(
586 "Deferred configuration apply completed but resulting config is invalid");
587 }
588
589 return;
590 }
591
592 // Publish device-status updates opportunistically when connected.
593 // Do at most one deferred operation per loop() pass.
594 if (pendingStatusPublish && Particle.connected()) {
595 if (writeDeviceStatusToCloud(pendingStatusPublishSource)) {
596 pendingStatusPublish = false;
597 pendingStatusPublishSource = "Unknown";
598 }
599 }
