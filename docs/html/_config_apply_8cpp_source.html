3bool Cloud::applyConfigurationFromLedger() {
4 bool success = true;
5
6 success &= applySensorConfig();
7 success &= applyTimingConfig();
8 success &= applyMessagingConfig();
9 success &= applyModesConfig();
10 success &= applyReportingConfig();
11
12 if (success) {
13 // Do not force synchronous storage flushes here; they can exceed the
14 // 100 ms loop budget. Persistence is handled by sysStatus.loop() and
15 // sensorConfig.loop() (called from the main loop).
16 sysStatus.validate(sizeof(sysStatus));
17 sensorConfig.validate(sizeof(sensorConfig));
18
19 // Defer device-status publishing to Cloud::loop() so it doesn't
20 // execute inside CONNECTING_STATE or async callbacks.
21 pendingStatusPublish = true;
22 } else {
23 Log.warn("Some configuration sections failed to apply");
24 }
25
26 return success;
27}
