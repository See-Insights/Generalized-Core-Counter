52void handleOccupancyMode() {
53 // Check if sensor has new data
54 if (SensorManager::instance().loop()) {
55 // Sensor detected presence
56 if (!current.get_occupied()) {
57 // Transition from unoccupied to occupied
58 current.set_occupied(true);
59 current.set_occupancyStartTime(Time.now());
60
61 // V3.23: Set LED with debounce timeout from sensor.setting1
62 uint32_t setting1Raw = sensorConfig.get_sensorSetting1();
63 uint32_t debounceSeconds = setting1Raw / 1000;
64 Log.info("OCCUPANCY detection: setting1=%lu ms => debounce=%lu sec",
65 (unsigned long)setting1Raw, (unsigned long)debounceSeconds);
66 if (debounceSeconds == 0) {
67 Log.warn("Debounce timeout is 0 - using 60 sec default (check sensor.setting1=%lu)",
68 (unsigned long)setting1Raw);
69 debounceSeconds = 60; // Minimum 60 second debounce for occupancy
70 }
71 signalLED(true, debounceSeconds * 1000UL); // Turn on LED until debounce expires
72
73 Log.info("Space now OCCUPIED at %s (LED timeout in %lu sec)", Time.timeStr().c_str(), debounceSeconds);
74
75 // In INTERMITTENT_KEEP_ALIVE mode, report immediately on occupancy state changes
76 // This allows dashboard to show real-time occupancy transitions
77 if (sysStatus.get_connectionMode() == INTERMITTENT_KEEP_ALIVE) {
78 session.occupancyChangeTriggered = true;
79 if (state == IDLE_STATE) {
80 Log.info("Occupancy change detected - triggering immediate report");
81 state = REPORTING_STATE;
82 }
83 }
84 } else {
85 // Already occupied - reset LED timeout on new motion
86 uint32_t debounceSeconds = sensorConfig.get_sensorSetting1() / 1000;
87 if (debounceSeconds == 0) {
88 debounceSeconds = 60; // Minimum 60 second debounce for occupancy
89 }
90 signalLED(true, debounceSeconds * 1000UL); // Reset LED timeout
91 if (sysStatus.get_verboseMode()) {
92 Log.info("Motion detected - LED timeout reset to %lu sec", debounceSeconds);
93 }
94 }
95
96 // Update last event time (resets debounce timer)
97 current.set_lastOccupancyEvent(millis());
98
99 if (sysStatus.get_verboseMode()) {
100 uint32_t occupiedDuration = Time.now() - current.get_occupancyStartTime();
101 Log.info("Occupancy event - Duration: %lu seconds", occupiedDuration);
102 }
103 }
104
105 // Check if we need to update occupancy state (timeout check)
106 updateOccupancyState();
107}

117void updateOccupancyState() {
118 if (!current.get_occupied()) {
119 return; // Nothing to do if not occupied
120 }
121
122 // V3.23: Occupancy debounce timeout comes from sensor.setting1
123 uint32_t debounceMs = sensorConfig.get_sensorSetting1();
124 if (debounceMs == 0) {
125 debounceMs = 60000; // default 60s
126 }
127
128 uint32_t lastEvent = current.get_lastOccupancyEvent();
129 if (lastEvent == 0) {
130 // If this is 0 (for example, occupancy set from a PIR wake path that
131 // didn't update lastOccupancyEvent), we must not immediately expire.
132 lastEvent = millis();
133 current.set_lastOccupancyEvent(lastEvent);
134 }
135 uint32_t timeSinceLastEvent = millis() - lastEvent;
136
137 // Check if debounce timeout has expired
138 if (timeSinceLastEvent > debounceMs) {
139 // Calculate this occupancy session duration
140 uint32_t sessionDuration = Time.now() - current.get_occupancyStartTime();
141
142 // Add to total occupied seconds for the day
143 uint32_t totalOccupied = current.get_totalOccupiedSeconds() + sessionDuration;
144 current.set_totalOccupiedSeconds(totalOccupied);
145
146 // Mark as unoccupied
147 current.set_occupied(false);
148 current.set_occupancyStartTime(0);
149
150 Log.info("Space now UNOCCUPIED - Session duration: %lu seconds, Total today: %lu seconds",
151 sessionDuration, totalOccupied);
152
153 signalLED(false); // Turn off LED
154
155 // In INTERMITTENT_KEEP_ALIVE mode, report immediately on occupancy state changes
156 // This allows dashboard to show real-time occupancy transitions
157 if (sysStatus.get_connectionMode() == INTERMITTENT_KEEP_ALIVE) {
158 session.occupancyChangeTriggered = true;
159 if (state == IDLE_STATE) {
160 Log.info("Occupancy change detected - triggering immediate report");
161 state = REPORTING_STATE;
162 }
163 }
