45ConnAcquirePhase classifyConnAcquirePhase(bool cellularReady,
46 bool networkReady,
47 bool cloudConnected) {
48 if (cloudConnected) {
49 return ConnAcquirePhase::CONNECTED;

51 if (!cellularReady) {
52 return ConnAcquirePhase::CELLULAR_ACQUIRE;
53 }
54 if (!networkReady) {
55 return ConnAcquirePhase::NETWORK_ACQUIRE;

57 return ConnAcquirePhase::CLOUD_ACQUIRE;

93struct ConnectBudgetContext {
94 unsigned long budgetMs = ConnectivityPolicy::CONNECT_BUDGET_DEFAULT_MS;
95 uint16_t configuredBudgetSec = 0;
96 uint8_t attemptCounter = 0;
97 float currentSoC = 0.0f;
98 bool allowDeepAttempt = false;

102ConnectBudgetContext evaluateConnectBudget() {
103 ConnectBudgetContext context;
104 context.configuredBudgetSec = sysStatus.get_connectAttemptBudgetSec();
105 if (context.configuredBudgetSec >= ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MIN_SEC &&
106 context.configuredBudgetSec <= ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MAX_SEC) {
107 context.budgetMs = (unsigned long)context.configuredBudgetSec * 1000UL;
108 } else if (context.configuredBudgetSec > 0 &&
109 context.configuredBudgetSec < ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MIN_SEC) {
110 context.configuredBudgetBelowMinimum = true;
111 context.budgetMs = ConnectivityPolicy::CONNECT_BUDGET_DEFAULT_MS;
112 }
113
114 context.attemptCounter = sysStatus.get_connectionAttemptCounter();
115 context.currentSoC = current.get_stateOfCharge();
116 context.allowDeepAttempt =
117 (context.attemptCounter >= ConnectivityPolicy::DEEP_ATTEMPT_COUNTER_THRESHOLD) ||
118 (context.currentSoC > ConnectivityPolicy::DEEP_ATTEMPT_SOC_THRESHOLD);
119 if (context.allowDeepAttempt) {
120 context.budgetMs = ConnectivityPolicy::CONNECT_BUDGET_DEEP_MS;
121 }
122
123 return context;
124}

714// FIRMWARE_UPDATE_STATE: Stay connected for firmware/config updates
715void handleFirmwareUpdateState() {
716 // Track how long we've been in update mode so we can mirror the
717 // Particle Wake-Publish-Sleep example behaviour: bound the time
718 // spent waiting for an update before going back to sleep.
719 static unsigned long firmwareUpdateStartMs = 0;
720 static bool configLoadedInUpdateMode = false;
721
722 if (state != oldState) {
723 publishStateTransition();
724 Log.info("Entering FIRMWARE_UPDATE_STATE - keeping device connected for updates");
725
726 firmwareUpdateStartMs = millis();
727 configLoadedInUpdateMode = false;
728
729 // Ensure cloud connection is requested
730 if (!Particle.connected()) {
731 Particle.connect();
732 }
733 }
734
735 // Once connected, ensure configuration is loaded at least once
736 if (Particle.connected()) {
737 if (!configLoadedInUpdateMode) {
738 Log.info("Connected in FIRMWARE_UPDATE_STATE - loading configuration from cloud");
739 Cloud::instance().loadConfigurationFromCloud();
740 configLoadedInUpdateMode = true;
741 }
742
743 // If no updates are pending anymore and no OTA in progress, exit update mode
744 if (!System.updatesPending()) {
745 Log.info("No updates pending - leaving FIRMWARE_UPDATE_STATE to IDLE_STATE");
746 configLoadedInUpdateMode = false;
747 transitionTo(IDLE_STATE, "firmware-update-complete");
748 return;
749 }
750 }
751
752 // Optional escape hatch: user button can also exit update mode
753 if (!digitalRead(BUTTON_PIN)) { // Active-low user button
754 Log.info("User button pressed - exiting FIRMWARE_UPDATE_STATE to IDLE_STATE");
755 transitionTo(IDLE_STATE, "firmware-update-button-exit");
756 return;
757 }
758
759 // Firmware update timeout: if we've spent more than firmwareUpdateMaxMs
760 // in this state, mirror the reference example and go to sleep so we can
761 // try again in a future cycle instead of keeping the modem on
762 // indefinitely.
763 if (firmwareUpdateStartMs != 0 && (millis() - firmwareUpdateStartMs) > firmwareUpdateMaxMs) {
764 Log.info("Firmware update timed out after %lu ms in FIRMWARE_UPDATE_STATE - transitioning to SLEEPING_STATE",
765 (unsigned long)(millis() - firmwareUpdateStartMs));
766 transitionTo(SLEEPING_STATE, "firmware-update-timeout");
767 }
