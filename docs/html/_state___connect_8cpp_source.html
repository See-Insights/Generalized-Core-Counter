32ConnectBudgetContext evaluateConnectBudget() {
33 ConnectBudgetContext context;
34 context.configuredBudgetSec = sysStatus.get_connectAttemptBudgetSec();
35 if (context.configuredBudgetSec >= ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MIN_SEC &&
36 context.configuredBudgetSec <= ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MAX_SEC) {
37 context.budgetMs = (unsigned long)context.configuredBudgetSec * 1000UL;
38 } else if (context.configuredBudgetSec > 0 &&
39 context.configuredBudgetSec < ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MIN_SEC) {
40 context.configuredBudgetBelowMinimum = true;
41 context.budgetMs = ConnectivityPolicy::CONNECT_BUDGET_DEFAULT_MS;
42 }
43
44 context.attemptCounter = sysStatus.get_connectionAttemptCounter();
45 context.currentSoC = current.get_stateOfCharge();
46 context.allowDeepAttempt =
47 (context.attemptCounter >= ConnectivityPolicy::DEEP_ATTEMPT_COUNTER_THRESHOLD) ||
48 (context.currentSoC > ConnectivityPolicy::DEEP_ATTEMPT_SOC_THRESHOLD);
49 if (context.allowDeepAttempt) {
50 context.budgetMs = ConnectivityPolicy::CONNECT_BUDGET_DEEP_MS;
51 }
52
53 return context;
54}

160 // ********** Connection Budget with Periodic Deep Attempts **********
161 // Per Particle cellular docs: Must allow at least 5 minutes for IMSI cycling,
162 // and periodically allow 11 minutes for full modem reset (occurs at 10 min mark).
163 // Base budget: 300s (5 minutes) - adequate for normal connection + IMSI cycling
164 // Deep budget: 660s (11 minutes) - allows full modem reset periodically
165
166 const ConnectBudgetContext budgetContext = evaluateConnectBudget();
167 budgetMs = budgetContext.budgetMs;
168 if (budgetContext.configuredBudgetBelowMinimum) {
169 Log.warn("Configured budget %us below 120s minimum, using 300s default",
170 budgetContext.configuredBudgetSec);
171 }
172
173 if (budgetContext.allowDeepAttempt) {
174 // Every 4th attempt (counter 0-3, resets at 3) OR when battery >50%,
175 // allow 11 minutes for full modem reset
176 if (budgetContext.attemptCounter >= ConnectivityPolicy::DEEP_ATTEMPT_COUNTER_THRESHOLD) {
177 Log.info("Deep connection attempt #%d - allowing 11 min for modem reset",
178 budgetContext.attemptCounter + 1);
179 sysStatus.set_connectionAttemptCounter(0); // Reset counter after deep attempt
180 } else {
181 Log.info("Healthy battery (%.1f%%) - allowing 11 min connection budget",
182 (double)budgetContext.currentSoC);
183 }
184 } else {
185 Log.trace("Normal connection attempt #%d - 5 min budget",
186 budgetContext.attemptCounter + 1);
187 }

189 armActiveConnectAttempt(budgetMs);

191 // Observability: record connect attempt type + final effective budget.
192 {
193 const uint16_t pending = (uint16_t)PublishQueuePosix::instance().getNumEvents();
194 Observability::cycleStats().markConnectAttempt(
195 budgetContext.allowDeepAttempt
196 ? Observability::WakeCycleStats::ConnectAttemptType::DEEP
197 : Observability::WakeCycleStats::ConnectAttemptType::NORMAL,
198 (uint32_t)budgetMs,
199 pending);
200 }
201 }

204 sysStatus.set_lastConnectionDuration(int(elapsedMs / 1000));
205
206 if (!connectRequested) {
207 // Log signal strength at start of connection attempt for field
208 // correlation with connectivity failures (alert 31). On cellular
209 // platforms this gives us a baseline RSSI before the modem fully
210 // connects, helping diagnose poor-reception issues.
211#if Wiring_Cellular

216 (void)qualityPct;
217#else
218#endif

222 setAppBreadcrumb(6);

232 // Heartbeat is emitted only while the connect attempt is active and still
233 // within the selected budget, so genuine over-budget stalls still timeout.

346 #endif

349 session.firstConnectionObserved = true;

379 (unsigned long)System.freeMemory());

385

388 clearActiveConnectAttempt();

393// FIRMWARE_UPDATE_STATE: Stay connected for firmware/config updates
394void handleFirmwareUpdateState() {
395 // Track how long we've been in update mode so we can mirror the
396 // Particle Wake-Publish-Sleep example behaviour: bound the time
397 // spent waiting for an update before going back to sleep.
398 static unsigned long firmwareUpdateStartMs = 0;
399
400 if (state != oldState) {
401 publishStateTransition();
402 Log.info("Entering FIRMWARE_UPDATE_STATE - keeping device connected for updates");
403
404 firmwareUpdateStartMs = millis();
405
406 // Ensure cloud connection is requested
407 if (!Particle.connected()) {
408 Particle.connect();
409 }
410 }
411
412 // Once connected, ensure configuration is loaded at least once
413 if (Particle.connected()) {
414 static bool configLoadedInUpdateMode = false;
415 if (!configLoadedInUpdateMode) {
416 Log.info("Connected in FIRMWARE_UPDATE_STATE - loading configuration from cloud");
417 Cloud::instance().loadConfigurationFromCloud();
418 configLoadedInUpdateMode = true;
419 }
420
421 // If no updates are pending anymore and no OTA in progress, exit update mode
422 if (!System.updatesPending()) {
423 Log.info("No updates pending - leaving FIRMWARE_UPDATE_STATE to IDLE_STATE");
424 configLoadedInUpdateMode = false;
425 state = IDLE_STATE;
426 return;
427 }
428 }
429
430 // Optional escape hatch: user button can also exit update mode
431 if (!digitalRead(BUTTON_PIN)) { // Active-low user button
432 Log.info("User button pressed - exiting FIRMWARE_UPDATE_STATE to IDLE_STATE");
433 state = IDLE_STATE;
434 return;
435 }
436
437 // Firmware update timeout: if we've spent more than firmwareUpdateMaxMs
438 // in this state, mirror the reference example and go to sleep so we can
439 // try again in a future cycle instead of keeping the modem on
440 // indefinitely.
441 if (firmwareUpdateStartMs != 0 && (millis() - firmwareUpdateStartMs) > firmwareUpdateMaxMs) {
442 Log.info("Firmware update timed out after %lu ms in FIRMWARE_UPDATE_STATE - transitioning to SLEEPING_STATE",
443 (unsigned long)(millis() - firmwareUpdateStartMs));
444 state = SLEEPING_STATE;
445 }
