63 // Per Particle cellular docs: Must allow at least 5 minutes for IMSI cycling,
64 // and periodically allow 11 minutes for full modem reset (occurs at 10 min mark).
65 // Base budget: 300s (5 minutes) - adequate for normal connection + IMSI cycling
66 // Deep budget: 660s (11 minutes) - allows full modem reset periodically

72 if (configuredBudgetSec >= ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MIN_SEC &&
73 configuredBudgetSec <= ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MAX_SEC) {
74 budgetMs = (unsigned long)configuredBudgetSec * 1000UL;
75 } else if (configuredBudgetSec > 0 && configuredBudgetSec < ConnectivityPolicy::CONNECT_BUDGET_CONFIG_MIN_SEC) {
76 Log.warn("Configured budget %us below 120s minimum, using 300s default", configuredBudgetSec);
77 budgetMs = ConnectivityPolicy::CONNECT_BUDGET_DEFAULT_MS;
78 }

82 float currentSoC = current.get_stateOfCharge();
83 bool allowDeepAttempt = (attemptCounter >= ConnectivityPolicy::DEEP_ATTEMPT_COUNTER_THRESHOLD) ||

92 (uint32_t)budgetMs,

94 }

235// FIRMWARE_UPDATE_STATE: Stay connected for firmware/config updates
236void handleFirmwareUpdateState() {
237 // Track how long we've been in update mode so we can mirror the
238 // Particle Wake-Publish-Sleep example behaviour: bound the time
239 // spent waiting for an update before going back to sleep.
240 static unsigned long firmwareUpdateStartMs = 0;
241
242 if (state != oldState) {
243 publishStateTransition();
244 Log.info("Entering FIRMWARE_UPDATE_STATE - keeping device connected for updates");
245
246 firmwareUpdateStartMs = millis();
247
248 // Ensure cloud connection is requested
249 if (!Particle.connected()) {
250 Particle.connect();
251 }
252 }
253
254 // Once connected, ensure configuration is loaded at least once
255 if (Particle.connected()) {
256 static bool configLoadedInUpdateMode = false;
257 if (!configLoadedInUpdateMode) {
258 Log.info("Connected in FIRMWARE_UPDATE_STATE - loading configuration from cloud");
259 Cloud::instance().loadConfigurationFromCloud();
260 configLoadedInUpdateMode = true;
261 }
262
263 // If no updates are pending anymore and no OTA in progress, exit update mode
264 if (!System.updatesPending()) {
265 Log.info("No updates pending - leaving FIRMWARE_UPDATE_STATE to IDLE_STATE");
266 configLoadedInUpdateMode = false;
267 state = IDLE_STATE;
268 return;
269 }
270 }
271
272 // Optional escape hatch: user button can also exit update mode
273 if (!digitalRead(BUTTON_PIN)) { // Active-low user button
274 Log.info("User button pressed - exiting FIRMWARE_UPDATE_STATE to IDLE_STATE");
275 state = IDLE_STATE;
276 return;
277 }
278
279 // Firmware update timeout: if we've spent more than firmwareUpdateMaxMs
280 // in this state, mirror the reference example and go to sleep so we can
281 // try again in a future cycle instead of keeping the modem on
282 // indefinitely.
283 if (firmwareUpdateStartMs != 0 && (millis() - firmwareUpdateStartMs) > firmwareUpdateMaxMs) {
284 Log.info("Firmware update timed out after %lu ms in FIRMWARE_UPDATE_STATE - transitioning to SLEEPING_STATE",
285 (unsigned long)(millis() - firmwareUpdateStartMs));
286 state = SLEEPING_STATE;
287 }
