198 // ********** Power Management **********
199 // In INTERMITTENT (1) or DISCONNECTED (2) modes, manage connection lifecycle.
200 if (sysStatus.get_connectionMode() != CONNECTED) {
201 // In CONNECTED mode during open hours, never auto-sleep.

208 // In low-power mode, once all work for this connection cycle is
209 // complete (no updates pending), we can safely enter SLEEPING_STATE
210 // to turn off the radio and save power. We only require the publish
211 // queue to be fully drained when we are actually connected; when
212 // offline, it's expected to have a non-zero queue and we still want
213 // to sleep, flushing the queue on the next connection.
214 bool canSleepGate = true;
215 if (Particle.connected()) {
216 canSleepGate = PublishQueuePosix::instance().getCanSleep();
217 }
218
219 if (!updatesPending && canSleepGate) {
220 // If a sensor event is still pending or the BLUE LED is still on
221 // from a recent count, defer transitioning into the SLEEPING_STATE.
222 // This avoids rapid Idle<->Sleeping ping-pong and the associated
223 // extra logging while still honouring the low-power policy once
224 // the indication has finished.
225 // NOTE: In OCCUPANCY mode, we intentionally allow sleep even while the
226 // debounce/LED timer is running. The PIR interrupt will wake us early
227 // for new motion, and SLEEPING_STATE already wakes on timer to evaluate
228 // the debounce timeout. Keeping the device awake here causes it to
229 // remain connected and burn power during continued motion.
230 if (sensorDetect || (sysStatus.get_sensorMode() == COUNTING && signalLEDTimeRemaining() > 0)) {
231 return;
232 }
233
234 size_t pending = PublishQueuePosix::instance().getNumEvents();
235 if (!Particle.connected() && pending > 0) {
236 Log.info("Low-power idle: offline with %u queued event(s) - sleeping and will flush on next connect",
237 (unsigned)pending);
238 } else {
239 Log.info("Low-power idle: queue drained and no updates pending - entering SLEEPING_STATE");
240 }
241 state = SLEEPING_STATE;
242 return; // Go back to sleep when there's no work this hour
243 }

246 // ********** IDLE Connectivity Ceiling Safety Net **********
247 // ThrashGuard intentionally does not supervise IDLE_STATE, so enforce a
248 // conservative max awake ceiling when cloud/modem remains powered without
249 // meaningful work. This prevents modem-on battery drain wedges.
250 {
251 static unsigned long idleCeilingStartMs = 0;
252
253 const unsigned long nowMs = millis();
254 const bool cloudConnected = Particle.connected();
255 const bool radioOn = Connectivity::isRadioPoweredOn();
256 const bool connectivityPowered = cloudConnected || radioOn;
257 const bool updatesPending = System.updatesPending();
258
259 bool queueCanSleep = true;
260 if (cloudConnected) {
261 queueCanSleep = PublishQueuePosix::instance().getCanSleep();
262 }
263
264 const bool openHoursKeepAwakeValid = Time.isValid() && isWithinOpenHours();
265 const bool healthyConnectedAwakePath =
266 (sysStatus.get_connectionMode() == CONNECTED) &&
267 openHoursKeepAwakeValid &&
268 cloudConnected;
269
270 const bool noMeaningfulWorkRemains = !updatesPending && queueCanSleep;
271 const bool shouldApplyIdleCeiling =
272 connectivityPowered &&
273 noMeaningfulWorkRemains &&
274 !healthyConnectedAwakePath;
275
276 if (!shouldApplyIdleCeiling) {
277 idleCeilingStartMs = 0;
278 } else {
279 // Reuse existing connectAttemptBudgetSec behavior with conservative fallback.
280 uint16_t budgetSec = sysStatus.get_connectAttemptBudgetSec();
281 if (budgetSec < 30 || budgetSec > 900) {
282 budgetSec = 300;
283 }
284 const unsigned long budgetMs = (unsigned long)budgetSec * 1000UL;
285
286 // Use connectedStartMs when available; otherwise track this IDLE-powered
287 // dwell with a local timer so radio-on/cloud-off wedges are also bounded.
288 const unsigned long startMs = (connectedStartMs != 0) ? connectedStartMs : idleCeilingStartMs;
289 if (startMs == 0) {
290 idleCeilingStartMs = nowMs;
291 } else {
292 const unsigned long elapsedMs = nowMs - startMs;
293 if (elapsedMs > budgetMs) {
294 Log.warn("IDLE ceiling trip: mode=%d cloud=%d radioOn=%d elapsedMs=%lu connectedStartMs=%lu -> forcing teardown and sleep",
295 (int)sysStatus.get_connectionMode(),
296 (int)cloudConnected,
297 (int)radioOn,
298 elapsedMs,
299 connectedStartMs);
300
301 Connectivity::requestFullDisconnectAndRadioOff();
302 connectedStartMs = 0;
303 idleCeilingStartMs = 0;
304 state = SLEEPING_STATE;
305 return;
306 }
307 }
