174PowerConfigurationApplyResult applyInputProfile(PowerInputProfile profile) {
175 PowerConfigurationApplyResult result;
176
177#if HAL_PLATFORM_CELLULAR && (PLATFORM_ID != PLATFORM_MSOM)
178 result.supported = true;
179
180 switch (profile) {

183 result.applied = (result.systemResult == SYSTEM_ERROR_NONE);
184 logAppliedProfileConfig(profile,

186 USB_BENCH_MIN_VOLTAGE_MV,
187 USB_BENCH_CHARGE_CURRENT_MA,
188 USB_BENCH_CHARGE_VOLTAGE_MV,
189 false,
190 result.systemResult);
191 return result;
192
193 case PowerInputProfile::Solar35W:
194 result.systemResult = System.setPowerConfiguration(makeSolarConfiguration());
195 result.applied = (result.systemResult == SYSTEM_ERROR_NONE);
196 logAppliedProfileConfig(profile,
197 SOLAR_MAX_CURRENT_MA,
198 SOLAR_MIN_VOLTAGE_MV,
199 SOLAR_CHARGE_CURRENT_MA,
200 SOLAR_CHARGE_VOLTAGE_MV,
201 true,
202 result.systemResult);
203 return result;
204
205 case PowerInputProfile::Auto:
206 case PowerInputProfile::NotApplicable:
207 result.systemResult = SYSTEM_ERROR_NONE;
208 return result;
209 }
210#else
211 (void)profile;
212#endif
213
214 result.systemResult = SYSTEM_ERROR_NONE;
215 return result;
