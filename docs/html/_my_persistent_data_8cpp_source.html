539void currentStatusData::resetEverything() { // The device is waking up in a new day or is a new install
540 current.set_lastCountTime(Time.now());
541 sysStatus.set_resetCount(0); // Reset the reset count as well
542
543 // ********** Reset Counting Mode Fields **********
544 current.set_hourlyCount(0);
545 current.set_dailyCount(0);
546
547 // ********** Reset Occupancy Mode Fields **********

554bool currentStatusData::validate(size_t dataSize) {
555 bool valid = PersistentDataFile::validate(dataSize);
556 if (valid) {
557 // Basic sanity checks on data
558 if (current.get_hourlyCount() > 10000 || current.get_dailyCount() > 100000) {
559 Log.info("Current: counts appear invalid, resetting");
560 current.set_hourlyCount(0);
561 current.set_dailyCount(0);
562 valid = false;
563 }
564
565 // Occupancy-mode sanity checks (prevents bogus 100+ year totals)
566 // totalOccupiedSeconds is "today" and should never exceed 24 hours.
567 uint32_t totalOccupied = current.get_totalOccupiedSeconds();
568 if (totalOccupied > 24UL * 3600UL) {
569 Log.warn("Current: totalOccupiedSeconds invalid (%lu sec) - resetting", (unsigned long)totalOccupied);
570 current.set_totalOccupiedSeconds(0);
571 }
572
573 // If occupied is true, occupancyStartTime must be non-zero and plausible.
574 if (current.get_occupied()) {
575 time_t start = current.get_occupancyStartTime();
576 if (start == 0) {
577 Log.warn("Current: occupied=true but occupancyStartTime=0 - forcing unoccupied");
578 current.set_occupied(false);
579 current.set_lastOccupancyEvent(0);
580 } else if (Time.isValid()) {
581 time_t now = Time.now();
582 // If start is in the future by more than a few seconds, clamp.
583 if (start > now + 5) {
584 Log.warn("Current: occupancyStartTime in future (%lu > %lu) - clamping", (unsigned long)start, (unsigned long)now);
585 current.set_occupancyStartTime(now);
586 }
587 }
588
589 // lastOccupancyEvent drives debounce logic; if missing, seed to now to avoid immediate expiry.
590 if (current.get_lastOccupancyEvent() == 0) {
591 current.set_lastOccupancyEvent(millis());
592 }
593 }
594 }
595 Log.info("Current data is %s", (valid) ? "valid" : "not valid");
596 return valid;
597}

682// Local helper to convert an alert code into a coarse severity bucket.
683// Higher numbers indicate more severe conditions.
684static int getAlertSeverity(int8_t code) {
685 if (code <= 0) {
686 return 0; // no alert
687 }
688
689 // Map known codes into tiers. This is intentionally simple and can be
690 // extended as new alert codes are added over time.
691 switch (code) {
692 case 14: // out-of-memory
693 case 15: // modem / disconnect failure
694 case 16: // repeated sleep failures (HIBERNATE / ULP / STOP)
695 case 17: // boot storm detected during setup/early boot
696 case 18: // state machine thrash detected (ThrashGuard)
697 case 20: // PMIC thermal shutdown (critical battery/charging fault)
698 case 21: // PMIC charge timeout / stuck charging
699 return 3; // critical
700
701 case 23: // PMIC battery fault (general)
702 case 30: // connectivity timeout with radio up
703 case 31: // failed to connect to cloud
704 case 32: // connect taking too long
705 case 40: // repeated webhook failures
