78bool sysStatusData::validate(size_t dataSize) {
79 bool valid = PersistentDataFile::validate(dataSize);
80 if (valid) {
81 // If test1 < 0 or test1 > 100, then the data is invalid
82
83 // openTime is an hour-of-day in local time (0-23)
84 if (sysStatus.get_openTime() > 23) {
85 Log.info("data not valid open time =%d", sysStatus.get_openTime());
86 valid = false;
87 }
88
89 if (sysStatus.get_closeTime() > 23) {
90 Log.info("data not valid close time =%d", sysStatus.get_closeTime());
91 valid = false;
92 }
93
94 // Last connection duration sanity check (seconds)
95 if (sysStatus.get_lastConnectionDuration() > 900) {
96 Log.info("data not valid last connection duration =%d", sysStatus.get_lastConnectionDuration());
97 valid = false;
98 }
99
100 if (sysStatus.get_connectivityRecoveryStage() > 3) {
101 Log.info("data not valid connectivity recovery stage =%d", sysStatus.get_connectivityRecoveryStage());
102 valid = false;
103 }
104
105 if (sysStatus.getValue<uint8_t>(offsetof(SysData,reservedByte0)) > 1) {
106 Log.info("data not valid enable hibernate flag");
107 valid = false;
108 }
109 }
110 if (!valid) {
111 Log.warn("sysStatus data is not valid");
112 }
113 return valid;

628 current.set_lastCountTime(Time.now());
629 sysStatus.set_resetCount(0); // Reset the reset count as well
630
631 // ********** Reset Counting Mode Fields **********
632 current.set_hourlyCount(0);
633 current.set_dailyCount(0);
634
635 // ********** Reset Occupancy Mode Fields **********
636 current.set_occupied(false);
637 current.set_lastOccupancyEvent(0);
638 current.set_occupancyStartTime(0);

642bool currentStatusData::validate(size_t dataSize) {
643 bool valid = PersistentDataFile::validate(dataSize);
644 if (valid) {
645 // Basic sanity checks on data
646 if (current.get_hourlyCount() > 10000 || current.get_dailyCount() > 100000) {
647 Log.info("Current: counts appear invalid, resetting");
648 current.set_hourlyCount(0);
649 current.set_dailyCount(0);
650 valid = false;
651 }
652
653 // Occupancy-mode sanity checks (prevents bogus 100+ year totals)
654 // totalOccupiedSeconds is "today" and should never exceed 24 hours.
655 uint32_t totalOccupied = current.get_totalOccupiedSeconds();
656 if (totalOccupied > 24UL * 3600UL) {
657 Log.warn("Current: totalOccupiedSeconds invalid (%lu sec) - resetting", (unsigned long)totalOccupied);
658 current.set_totalOccupiedSeconds(0);
659 }
660
661 // If occupied is true, occupancyStartTime must be non-zero and plausible.
662 if (current.get_occupied()) {
663 time_t start = current.get_occupancyStartTime();
664 if (start == 0) {
665 Log.warn("Current: occupied=true but occupancyStartTime=0 - forcing unoccupied");
666 current.set_occupied(false);
667 current.set_lastOccupancyEvent(0);
668 } else if (Time.isValid()) {
669 time_t now = Time.now();
670 // If start is in the future by more than a few seconds, clamp.
671 if (start > now + 5) {
672 Log.warn("Current: occupancyStartTime in future (%lu > %lu) - clamping", (unsigned long)start, (unsigned long)now);
673 current.set_occupancyStartTime(now);
674 }
675 }
676
677 // lastOccupancyEvent drives debounce logic; if missing, seed to now to avoid immediate expiry.
678 if (current.get_lastOccupancyEvent() == 0) {
679 current.set_lastOccupancyEvent(millis());
680 }
681 }
682 }

772// Local helper to convert an alert code into a coarse severity bucket.
773// Higher numbers indicate more severe conditions.
774static int getAlertSeverity(int8_t code) {
775 if (code <= 0) {
776 return 0; // no alert
777 }
778
779 // Map known codes into tiers. This is intentionally simple and can be
780 // extended as new alert codes are added over time.
781 switch (code) {
782 case 14: // out-of-memory
783 case 15: // modem / disconnect failure
784 case 16: // repeated sleep failures (HIBERNATE / ULP / STOP)
785 case 17: // boot storm detected during setup/early boot
786 case 18: // state machine thrash detected (ThrashGuard)
787 case 20: // PMIC thermal shutdown (critical battery/charging fault)
788 case 21: // PMIC charge timeout / stuck charging
789 return 3; // critical
790
791 case 23: // PMIC battery fault (general)
792 case 30: // connectivity timeout with radio up
793 case 31: // failed to connect to cloud
794 case 32: // connect taking too long
795 case 40: // repeated webhook failures
796 case 41: // configuration/ledger apply failure (CONNECT phase)
797 case 42: // data ledger publish failure
798 case 43: // publish queue not drained before forced sleep
799 return 2; // major
800 case 44: // ledger sync timeout before sleep (SLEEP phase - cosmetic, config already applied)
801 return 1; // minor - less severe than 41 because config is already working
802 default:
803 return 1; // minor / warning
804 }
805}
806
807void currentStatusData::raiseAlert(int8_t value) {
808 if (value <= 0) {
809 return; // ignore attempts to "raise" a non-alert here
810 }
811
812 int8_t existing = get_alertCode();
813 if (getAlertSeverity(value) > getAlertSeverity(existing)) {
814 set_alertCode(value);
815 set_lastAlertTime(Time.now());
816 }
817}
