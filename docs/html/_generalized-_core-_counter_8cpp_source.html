60/*
61 * Architectural overview
62 * ----------------------
63 * - State machine: setup()/loop() implement a simple state machine
64 * (INITIALIZATION, CONNECTING, IDLE, SLEEPING, REPORTING, ERROR)
65 * that drives sensing, reporting, and power management.
66 * - Sensor abstraction: ISensor + SensorFactory + SensorManager allow
67 * different physical sensors (PIR, ultrasonic, etc.) behind one API.
68 * - Cloud configuration: the Cloud singleton uses Particle Ledger to
69 * merge product defaults (default-settings) with per-device overrides
70 * (device-settings), then applies the merged config to persistent data.
71 * - Data publishing: publishData() builds a JSON payload and sends it
72 * via PublishQueuePosix (webhook) and also updates the device-data
73 * ledger for Console visibility.
74 * - Connectivity: compile-time macros (Wiring_WiFi / Wiring_Cellular)
75 * select WiFi vs. cellular for radio control; Particle.connect() is
76 * used to bring up the cloud session on both.
77 */

463void loop() {
464 // If we remain alive for a stability window, clear boot-storm counters.
465 if (bootStormCount > 0 && millis() > 300000UL) {
466 bootStormCount = 0;
467 bootStormWindowStart = 0;
468 }
469
470 // Main state machine driving sensing, reporting, power management
471 switch (state) {
472 case IDLE_STATE:
473 handleIdleState();
474 break;
475
476 case SLEEPING_STATE:
477 handleSleepingState();
478 break;
479
480 case REPORTING_STATE:
481 handleReportingState();
482 break;
483
484 case CONNECTING_STATE:
485 handleConnectingState();
486 break;
487
488 case FIRMWARE_UPDATE_STATE:
489 handleFirmwareUpdateState();
490 break;
491
492 case ERROR_STATE:
493 handleErrorState();
494 break;
495 }
496
497 thrashGuard.loop(state, millis());
498
499 ab1805.loop(); // Keeps the RTC synchronized with the device clock
500
501 // Housekeeping for each transit of the main loop
502 current.loop();
503 sysStatus.loop();
504 sensorConfig.loop();
505
506 // Service deferred cloud work (ledger status publishes, etc.)
507 Cloud::instance().loop();
508
509 // Service outgoing publish queue
510 PublishQueuePosix::instance().loop();
511
512 // Check for short-term webhook response timeout.
513 // Requirement: 20 seconds starting only after a successful cloud connect.
514 if (Particle.connected() && session.awaitingWebhookResponse && session.webhookAwaitStartMs != 0) {
515 unsigned long timeoutMs = sysStatus.get_webhookTimeoutMs();
516 if (timeoutMs < 5000UL || timeoutMs > 120000UL) {
517 timeoutMs = 20000UL;
518 }
519
520 const unsigned long elapsedMs = millis() - session.webhookAwaitStartMs;
521 if (elapsedMs > timeoutMs) {
522 Log.warn("Webhook response timeout after %lu ms - raising alert 40", elapsedMs);
523 session.awaitingWebhookResponse = false;
524 session.webhookAwaitStartMs = 0;
525 current.raiseAlert(40);
526 }
527 }
528
529 // If an out-of-memory event occurred, go to error state
530 if (outOfMemory >= 0) {
531 Log.info("Resetting due to low memory");
532 // Out-of-memory is treated as a critical alert; only overwrite any
533 // existing alert if this is more severe.
534 current.raiseAlert(14);
535 state = ERROR_STATE;
536 }
537
538 // If the user switch is pressed, force an immediate report and connection
539 if (userSwitchDetected) {
540 Log.info("User switch pressed - triggering immediate report and connection");
541 userSwitchDetected = false;
542 session.occupancyChangeTriggered = true; // Bypass boundary alignment for immediate connection
543 state = REPORTING_STATE;
544 }
545
546 // ********** Centralized sensor event handling **********
547 // Service sensor interrupts regardless of current state. This ensures
548 // counts are captured even during long-running operations like cellular
549 // connection attempts (which can take minutes) or firmware updates.
550 // MEASUREMENT mode is time-based (handled in IDLE only), not interrupt-driven.
551 uint8_t sensorMode = sysStatus.get_sensorMode();
552 if (sensorMode == COUNTING) {
553 handleCountingMode(); // Count each sensor event
554 } else if (sensorMode == OCCUPANCY) {
555 handleOccupancyMode(); // Track occupied/unoccupied state
556 }
557
558} // End of loop

567static bool isWithinOpenHoursForHour(uint8_t hour, uint8_t openHour, uint8_t closeHour) {
568 if (openHour < closeHour) {
569 // Simple daytime window, e.g. 6 -> 22
570 return (hour >= openHour) && (hour < closeHour);
571 } else if (openHour > closeHour) {
572 // Overnight window, e.g. 20 -> 6
573 return (hour >= openHour) || (hour < closeHour);
574 } else {
575 // openHour == closeHour: treat as always open
576 return true;
577 }
578}

580static int secondsUntilNextOpenForSeconds(uint32_t secondsOfDay,
581 uint8_t openHour,
582 uint8_t closeHour,
583 bool openNow) {
584 uint32_t openSec = (uint8_t)openHour * 3600;
585 uint32_t closeSec = (uint8_t)closeHour * 3600;
586
587 // Normalize: if we're currently within opening hours, next open is tomorrow
588 if (openNow) {
589 return (int)((24 * 3600UL - secondsOfDay) + openSec);
590 }
591
592 if (openHour < closeHour) {
593 // Simple daytime window, closed before open or after close
594 if (secondsOfDay < openSec) {
595 // Before opening today
596 return (int)(openSec - secondsOfDay);
597 } else {
598 // After closing, next open is tomorrow
599 return (int)((24 * 3600UL - secondsOfDay) + openSec);
600 }
601 } else if (openHour > closeHour) {
602 // Overnight window; closed between closeHour and openHour
603 if (secondsOfDay < openSec && secondsOfDay >= closeSec) {
604 // During the closed gap today
605 return (int)(openSec - secondsOfDay);
606 } else {
607 // Otherwise next open is later today or tomorrow, but openNow
608 // was already false so this path will generally be rare; fall back to 1 hour
609 return 3600;
610 }
611 } else {
612 // openHour == closeHour: always open; should not normally reach here
613 return 3600;
614 }
615}

618// Helper to determine whether current *local* time is within park open hours.
619// Local time is derived from LocalTimeRK using the configured timezone.
620// If time is not yet valid, we treat it as "open" so the device can start
621// sensing while it acquires time and configuration.
622bool isWithinOpenHours() {
623 if (!Time.isValid()) {
624 return true;
625 }
626
627 uint8_t openHour = sysStatus.get_openTime();
628 uint8_t closeHour = sysStatus.get_closeTime();
629 const LocalTimeCache::LocalTimeSnapshot &snapshot = LocalTimeCache::getLocalTimeSnapshot();
630 const uint8_t hour = snapshot.localHour;
631 const bool openNow = isWithinOpenHoursForHour(hour, openHour, closeHour);
632
633 return openNow;
634}

636// Helper to compute seconds until next park opening time (local time)
637int secondsUntilNextOpen() {
638 if (!Time.isValid()) {
639 // Fallback: 1 hour if time is not yet valid
640 return 3600;
641 }

643 uint8_t openHour = sysStatus.get_openTime();

666void publishData() {
667 // Legacy Ubidots context strings describing battery state
668 static const char *batteryContext[7] = {
669 "Unknown", "Not Charging", "Charging",
670 "Charged", "Discharging", "Fault",
671 "Diconnected"
672 };
673
674 char data[256];
675
676 // We support two timestamp strategies:
677 // - Occupancy mode: use current time so the value updates in Ubidots even
678 // if multiple publishes happen within the same hour.
679 // - Counting mode: keep the legacy "end of previous hour" timestamp so
680 // hourly bucket aggregation behavior remains unchanged.
681 const unsigned long nowStampSec = Time.now();
682 const unsigned long endOfPrevHourStampSec = nowStampSec - (Time.minute() * 60L + Time.second() + 1L);
683
684 // Bounds check battery state index for safety
685 uint8_t battState = current.get_batteryState();
686 if (battState > 6) {
687 battState = 0;
688 }
689
690 uint8_t sensorMode = sysStatus.get_sensorMode();
691
692 // Ensure battery value is always a finite number for webhook ingestion.
693 // Ubidots rejects NaN/Inf with a 400 response.
694 float stateOfCharge = current.get_stateOfCharge();
695 if (!(stateOfCharge == stateOfCharge) || stateOfCharge < 0.0f || stateOfCharge > 100.0f) {
696 stateOfCharge = 0.0f;
697 battState = 0;
698 }
699
700 // Build webhook payload based on sensor mode
701 if (sensorMode == OCCUPANCY) {
702 const unsigned long timeStampValue = nowStampSec;
703 const unsigned long totalOccupiedMinutes = (unsigned long)(current.get_totalOccupiedSeconds() / 60UL);
704
705 // Occupancy mode webhook format (occupancy as 0/1 numeric value)
706 snprintf(data, sizeof(data),
707 "{\"occupancy\":%d,\"dailyoccupancy\":%lu,\"battery\":%4.2f,\"key1\":\"%s\",\"temp\":%4.2f,\"alerts\":%i,\"resets\":%i,\"connecttime\":%i,\"timestamp\":%lu000}",
708 current.get_occupied() ? 1 : 0,
709 totalOccupiedMinutes,
710 stateOfCharge,
711 batteryContext[battState],
712 current.get_internalTempC(),
713 current.get_alertCode(),
714 sysStatus.get_resetCount(),
715 sysStatus.get_lastConnectionDuration(),
716 timeStampValue);
717
718 // Log occupancy state and total minutes
719 Log.info("Report payload: occupancy=%d totalMinutes=%lu alert=%d",
720 current.get_occupied() ? 1 : 0,
721 totalOccupiedMinutes,
722 (int)current.get_alertCode());
723 } else {
724 const unsigned long timeStampValue = endOfPrevHourStampSec;
725
726 // Counting mode webhook format (original format)
727 snprintf(data, sizeof(data),
728 "{\"hourly\":%i,\"daily\":%i,\"battery\":%4.2f,\"key1\":\"%s\",\"temp\":%4.2f,\"resets\":%i,\"alerts\":%i,\"connecttime\":%i,\"timestamp\":%lu000}",
729 current.get_hourlyCount(),
730 current.get_dailyCount(),
731 stateOfCharge,
732 batteryContext[battState],
733 current.get_internalTempC(),
734 sysStatus.get_resetCount(),
735 current.get_alertCode(),
736 sysStatus.get_lastConnectionDuration(),
737 timeStampValue);
738
739 // Explicitly log the counts and alert code used in this report
740 Log.info("Report payload: hourly=%d daily=%d alert=%d",
741 (int)current.get_hourlyCount(),
742 (int)current.get_dailyCount(),
743 (int)current.get_alertCode());
744 }
745
746 // Get webhook name from cloud configuration (with fallback to convention)
747 String webhookName = Cloud::instance().getWebhookName();
748
749 PublishQueuePosix::instance().publish(webhookName, data, PRIVATE);
750 Log.info("Publishing to webhook '%s': %s", webhookName.c_str(), data);
751
752 // Arm short-term webhook supervision.
753 // If we're already connected, start the 20s window immediately.
754 // Otherwise start it when CONNECTING_STATE reports a successful cloud connect.
755 session.webhookExpectedOnConnect = true;
756 if (Particle.connected()) {
757 session.webhookExpectedOnConnect = false;
758 session.awaitingWebhookResponse = true;
759 session.webhookAwaitStartMs = millis();
760 Log.info("Webhook queued while connected - awaiting response (short-term window started)");
761 }
762
763 // Also update device-data ledger with structured JSON snapshot
764 if (!Cloud::instance().publishDataToLedger()) {
765 // Data ledger publish failure; escalate via alert so the error
766 // supervisor can decide on corrective action.
767 current.raiseAlert(42);
768 }
769}

819void UbidotsHandler(const char *event, const char *data) {
820 // Handle response from Ubidots webhook (legacy integration)
821 char responseString[64];
822 // Response is expected to be a single numeric code from the Particle
823 // integration response template (e.g. "200" or "201").
824 if (!data || !strlen(data)) {
825 snprintf(responseString, sizeof(responseString), "No Data");
826 } else {
827 // Any webhook response indicates the integration path is alive.
828 // Update lastHookResponse even if it arrived after our short-term window.
829 sysStatus.set_lastHookResponse(Time.now());
830
831 if (session.awaitingWebhookResponse) {
832 session.awaitingWebhookResponse = false;
833 session.webhookAwaitStartMs = 0;
834 }
835
836 // Clear webhook supervision alert (40) on any response.
837 if (current.get_alertCode() == 40) {
838 current.set_alertCode(0);
839 current.set_lastAlertTime(0);
840 }
841
842 // If the response is numeric, treat 200/201 as success.
843 // Otherwise, treat it as an opaque success marker.
844 if (isdigit((unsigned char)data[0])) {
845 int code = atoi(data);
846 if (code == 200 || code == 201) {
847 snprintf(responseString, sizeof(responseString), "Response Received");
848 } else {
849 snprintf(responseString, sizeof(responseString), "Hook response %d", code);
850 }
851 } else {
852 snprintf(responseString, sizeof(responseString), "Response Received");
853 }
854 }
855 if (sysStatus.get_verboseMode() && Particle.connected()) {
856 publishDiagnosticSafe("Ubidots Hook", responseString, PRIVATE);
857 }
858 Log.info("%s", responseString);
859}

875bool publishDiagnosticSafe(const char* eventName, const char* data, PublishFlags flags) {
876 // Guard: only add diagnostics when queue has capacity for them.
877 // Reserve headroom for critical data payloads (hourly reports, alerts).
878 // Threshold: allow diagnostics if queue has <10 events pending.
879 const size_t DIAGNOSTIC_QUEUE_THRESHOLD = 10;
880
881 size_t queueDepth = PublishQueuePosix::instance().getNumEvents();
882
883 if (queueDepth >= DIAGNOSTIC_QUEUE_THRESHOLD) {
884 Log.info("Diagnostic publish skipped (queue depth=%u): %s", (unsigned)queueDepth, eventName);
885 return false;
886 }
887
888 // Queue has capacity; safe to add diagnostic message
889 PublishQueuePosix::instance().publish(eventName, data, flags);
890 return true;
891}

943void dailyCleanup() {
944 if (Particle.connected()) {
945 publishDiagnosticSafe("Daily Cleanup", "Running", PRIVATE);
946
947 // Force time sync once per day to prevent clock drift.
948 // dailyCleanup() is called once per day from REPORTING_STATE. All devices
949 // should connect at least daily, but LOW_POWER and DISCONNECTED mode devices
950 // may not be connected at the specific time dailyCleanup runs. If a device
951 // connects daily but misses the cleanup window each day, it won't sync time
952 // despite daily connections, causing drift to accumulate over multiple days.
953 // The AB1805 RTC has ±2.0 ppm accuracy (~±5 seconds/month typical), so
954 // missing sync for several days can accumulate noticeable drift. This ensures
955 // at least one time sync per day when the device happens to be connected
956 // during the cleanup window.
957 Log.info("Daily time sync requested");
