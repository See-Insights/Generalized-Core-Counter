69/*
70 * Architectural overview
71 * ----------------------
72 * - State machine: setup()/loop() implement a simple state machine
73 * (INITIALIZATION, CONNECTING, IDLE, SLEEPING, REPORTING, ERROR)
74 * that drives sensing, reporting, and power management.
75 * - Sensor abstraction: ISensor + SensorFactory + SensorManager allow
76 * different physical sensors (PIR, ultrasonic, etc.) behind one API.
77 * - Cloud configuration: the Cloud singleton uses Particle Ledger to
78 * merge product defaults (default-settings) with per-device overrides
79 * (device-settings), then applies the merged config to persistent data.
80 * - Data publishing: publishData() builds a JSON payload and sends it
81 * via PublishQueuePosix (webhook) and also updates the device-data
82 * ledger for Console visibility.
83 * - Connectivity: compile-time macros (Wiring_WiFi / Wiring_Cellular)
84 * select WiFi vs. cellular for radio control; Particle.connect() is
85 * used to bring up the cloud session on both.
86 *
87 * File navigation
88 * ---------------
89 * 1) Includes, globals, retained state, and small formatting helpers
90 * 2) setup()/loop() and top-level lifecycle wiring
91 * 3) policy helpers for battery tier, open hours, and report publishing
92 * 4) startup status, webhook supervision, diagnostics, and state logging
93 * 5) connectivity failsafe, ISRs, and daily maintenance
94 */

241uint32_t connectivityFailsafeJitterSec(uint8_t stage) {
242#if CONNECTIVITY_FAILSAFE_TEST_MODE
243 (void)stage;
244 return 0;
245#else
246 if (ConnectivityPolicy::CONNECTIVITY_FAILSAFE_JITTER_MAX_SEC <= 0) {
247 return 0;
248 }
249
250 String deviceId = System.deviceID();
251 uint32_t hash = 5381u;
252 for (size_t index = 0; index < deviceId.length(); ++index) {
253 hash = ((hash << 5) + hash) ^ (uint8_t)deviceId.charAt(index);
254 }
255 hash ^= (uint32_t)stage * 2654435761u;
256 return hash % (uint32_t)ConnectivityPolicy::CONNECTIVITY_FAILSAFE_JITTER_MAX_SEC;
257#endif
258}

291bool claimFailsafeDeferLogLocal(FailsafeDeferReason reason) {
292 if (reason == FAILSAFE_DEFER_NONE) {
293 return false;
294 }
295
296 const uint8_t shift = (uint8_t)reason - 1;
297 const uint32_t bit = (shift < 32) ? (1UL << shift) : 0UL;
298 if (bit == 0UL) {
299 return false;
300 }
301 if (session.failsafeDeferLogMask & bit) {
302 return false;
303 }
304 session.failsafeDeferLogMask |= bit;
305 return true;

358FailsafeDeferReason currentFailsafeEligibilityReason() {
359 if (sysStatus.get_connectionMode() == DISCONNECTED) {
360 return FAILSAFE_DEFER_DISCONNECTED_MODE;
361 }
362
363 if (state == FIRMWARE_UPDATE_STATE || System.updatesPending()) {
364 return FAILSAFE_DEFER_UPDATE_PENDING;
365 }
366
367 if (!Time.isValid()) {
368 return FAILSAFE_DEFER_INVALID_TIME;
369 }
370
371 const time_t lastConnection = sysStatus.get_lastConnection();
372 if (lastConnection == 0) {
373 return FAILSAFE_DEFER_NO_LAST_CONNECTION;
374 }
375
376 const time_t now = Time.now();
377 if (now > lastConnection) {
378 const time_t connectionAgeSec = now - lastConnection;
379 uint8_t currentStage = sysStatus.get_connectivityRecoveryStage();
380 if (currentStage > 3) {
381 currentStage = 0;
382 }
383
384 if (connectionAgeSec >= ConnectivityPolicy::CONNECTIVITY_FAILSAFE_STALE_SEC && currentStage < 3) {
385 time_t lastAction = sysStatus.get_lastConnectivityRecoveryAction();
386 if (lastAction > now) {
387 lastAction = 0;
388 }
389
390 const uint8_t nextStage = currentStage + 1;
391 time_t requiredDelay = ConnectivityPolicy::CONNECTIVITY_FAILSAFE_COOLDOWN_SEC;
392 if (nextStage >= 2) {
393 requiredDelay += (time_t)connectivityFailsafeJitterSec(nextStage);
394 }
395
396 const bool cooldownComplete =
397 (currentStage == 0 || lastAction == 0 || (now - lastAction) >= requiredDelay);
398
399 if (cooldownComplete) {
400 const BatteryTier tier = currentBatteryTierForFailsafe();
401 const bool externalPowerPresent = connectivityFailsafeHasExternalPower();
402 const bool lowBatteryHardActionBlocked =
403 nextStage >= 2 && !externalPowerPresent &&
404 (sysStatus.get_lowBatteryMode() || tier == TIER_SURVIVAL);
405
406 if (lowBatteryHardActionBlocked) {
407 return FAILSAFE_DEFER_LOW_BATTERY_HARD_STAGE_SUPPRESSED;
408 }
409 }
410 }
411 }
412
413 return FAILSAFE_DEFER_NONE;
414}

504void setup() {
505 const int reason = System.resetReason();
506 const uint32_t reasonData = System.resetReasonData();
507 const uint8_t previousBreadcrumb = appBreadcrumb;
508 const uint32_t previousBreadcrumbMs = appBreadcrumbMs;
509 startupPreviousBreadcrumb = previousBreadcrumb;
510 startupPreviousBreadcrumbMs = previousBreadcrumbMs;
511 setAppBreadcrumb(BREADCRUMB_SETUP_START);
512 bootStormLastResetReason = (uint8_t)reason;
513 bootStormLastResetReasonData = reasonData;
514
515 const bool previousBootFailedEarly = bootInProgress;
516 // Mark this boot as in-progress immediately so any early reset on this pass
517 // is detectable on the next boot.
518 bootInProgress = true;
519
520 bool qualifiesForStormCount = false;
521 switch (reason) {
522 case RESET_REASON_NONE:
523#ifdef RESET_REASON_PANIC
524 case RESET_REASON_PANIC:
525#endif
526 case RESET_REASON_WATCHDOG:
527#ifdef RESET_REASON_UNKNOWN
528 case RESET_REASON_UNKNOWN:
529#endif
530#ifdef RESET_REASON_USER_APPLICATION
531 case RESET_REASON_USER_APPLICATION:
532#endif
533 qualifiesForStormCount = true;
534 break;
535 default:
536 break;
537 }
538
539 if (previousBootFailedEarly && qualifiesForStormCount) {
540 Log.warn("Previous boot reset early before setup completed (reason=%d data=%lu)",
541 reason,
542 (unsigned long)reasonData);
543 }
544
545 if (previousBootFailedEarly && qualifiesForStormCount) {
546 if (bootStormCount < 255) {
547 bootStormCount++;
548 }
549 }
550
551 if (Time.isValid()) {
552 const time_t now = Time.now();
553 if (bootStormWindowStart == 0) {
554 bootStormWindowStart = now;
555 } else if ((now - bootStormWindowStart) > 600) {
556 bootStormCount = (previousBootFailedEarly && qualifiesForStormCount) ? 1 : 0;
557 bootStormWindowStart = now;
558 }
559 }
560
561 if (bootStormCount >= 6 && previousBootFailedEarly && qualifiesForStormCount) {
562 bootStormTripCount++;
563 bootStormAlertPending = true;
564 Log.error("BOOT STORM: %u early resets detected (reason=%d)", bootStormCount, reason);
565 Connectivity::requestFullDisconnectAndRadioOff();
566 Particle.process();
567 SystemSleepConfiguration bootStormSleep;
568 bootStormSleep.mode(SystemSleepMode::ULTRA_LOW_POWER).duration(600000UL);
569 System.sleep(bootStormSleep);
570 Log.warn("BOOT STORM holdoff sleep returned unexpectedly - continuing boot");
571 }
572
573 // Wait for serial connection only in explicit DEV builds
574#if ALLOW_BLOCKING_SERIAL_WAITS
575 waitFor(Serial.isConnected, 10000);
576 delay(1000);
577#endif
578
579 // Observability: start a new wake cycle on cold boot.
580 Observability::cycleStats().resetOnWake(millis());
581
582 System.on(out_of_memory,
583 outOfMemoryHandler); // Enabling an out of memory handler is a good
584 // safety tip. If we run out of memory a
585 // System.reset() is done.
586
587 // Application watchdog: reset if loop() doesn't execute within 60 seconds.
588 // This catches state machine hangs, blocking operations, and cellular/cloud
589 // stalls that exceed our non-blocking design intent. The AB1805 hardware
590 // watchdog (124s) provides ultimate backstop if this software watchdog fails.
591 static ApplicationWatchdog appWatchdog(60000, appWatchdogHandler, 1536);
592
593 // Subscribe to the Ubidots integration response event so we can track
594 // successful webhook deliveries and update lastHookResponse.
595 // This subscription is non-blocking; we'll receive responses when connected.
596 {
597 char responseTopic[125];
598 String deviceID = System.deviceID();
599 deviceID.toCharArray(responseTopic, sizeof(responseTopic));
600 Particle.subscribe(responseTopic, UbidotsHandler);
601
602 // Also subscribe to the default Particle webhook response prefix so this
603 // works with whatever webhook name is configured in the ledger.
604 // If the integration uses the default response topic, responses will be
605 // published to: hook-response/<eventName>.
606 Particle.subscribe("hook-response/", UbidotsHandler);
607 }
608
609 // Configure startup with the radio left off. CONNECTING_STATE owns
610 // Particle.connect() and all subsequent network bring-up.
611#if !Wiring_WiFi && !Wiring_Cellular
612 // Platforms without explicit Wiring_* radio macros still connect through CONNECTING_STATE.
613#endif
614
615 initializePinModes(); // Initialize the pin modes
616
617 // Recovery path: if the user holds the service button during reset/wake,
618 // force a cloud connection regardless of opening-hours logic.
619 // This allows ledger changes (e.g., opening hours) to be pulled down to
620 // recover a device that would otherwise immediately sleep.
621 if (digitalRead(BUTTON_PIN) == LOW) { // Active-low user button
622 Log.info("Boot override: user button held - forcing CONNECTING_STATE");
623 state = CONNECTING_STATE;
624 }
625
626 sysStatus.setup(); // Initialize persistent storage
627 sensorConfig.setup(); // Initialize the sensor configuration
628 current.setup(); // Initialize the current status data
629 PowerManager::instance().setup();
630
631 // If a boot storm holdoff was triggered on this or the prior boot, surface
632 // it as an alert now that persistent current status storage is initialized.
633 if (bootStormAlertPending) {
634 // Force explicit boot-storm alert visibility in startup/report payloads.
635 current.set_alertCode(17);
636 current.set_lastAlertTime(Time.now());
637 bootStormAlertPending = false;
638 Log.warn("Boot storm holdoff detected - raising alert 17");
639 }
640
641 // Configure serial logging based on serial flag
642 // Note: Particle firmware on this platform doesn't support Log.level()
643 if (sysStatus.get_serialConnected() || (ALLOW_BLOCKING_SERIAL_WAITS != 0)) {
644 Serial.begin(9600);
645
646 // Honor the persisted serialConnected flag with a bounded wait so
647 // USB serial can attach before a low-power device drops back to sleep.
648 // In developer builds, the build profile can force this wait even when
649 // cloud config has not enabled serial logging yet.
650 const unsigned long serialWaitStart = millis();
651 while (!Serial.isConnected() && (millis() - serialWaitStart) < ConnectivityPolicy::DEBUG_SERIAL_WAIT_TIMEOUT_MS) {
652 Particle.process();
653 delay(ConnectivityPolicy::DEBUG_SERIAL_WAIT_POLL_DELAY_MS);
654 }
655
656 if (Serial.isConnected()) {
657 delay(ConnectivityPolicy::DEBUG_SERIAL_POST_CONNECT_DELAY_MS);
658 }
659 }
660
661 if (sysStatus.get_testConnectionDurationOverride() == 0) {
662 sysStatus.set_testConnectionDurationOverride(0xFFFF); // Disabled (max uint16_t)

891 case SLEEPING_STATE:

893 break;
894
895 case REPORTING_STATE:

901 break;
902

904 handleFirmwareUpdateState();

908 handleErrorState();
909 break;

917 current.loop();
918 sysStatus.loop();
919 sensorConfig.loop();
920

922 Cloud::instance().loop();
923

925 PublishQueuePosix::instance().loop();

928 // Requirement: 20 seconds starting only after a successful cloud connect.
929 if (Particle.connected() && session.awaitingWebhookResponse && session.webhookAwaitStartMs != 0) {
930 unsigned long timeoutMs = sysStatus.get_webhookTimeoutMs();

970 handleCountingMode(); // Count each sensor event
971 } else if (sensorMode == OCCUPANCY) {
972 handleOccupancyMode(); // Track occupied/unoccupied state
973 }

975} // End of loop

1017 tierNames[newTier],
1018 (double)currentSoC);
1019 sysStatus.set_lowBatteryMode(false);
1020 }
1021 } else if (sysStatus.get_lowBatteryMode()) {
1022 Log.info("Battery recovery: clearing lowBatteryMode (tier=%s, SoC=%.1f%%)",
1023 tierNames[newTier],

1026 }

1031static bool isWithinOpenHoursForHour(uint8_t hour, uint8_t openHour, uint8_t closeHour) {
1032 if (openHour < closeHour) {
1033 // Simple daytime window, e.g. 6 -> 22
1034 return (hour >= openHour) && (hour < closeHour);
1035 } else if (openHour > closeHour) {
1036 // Overnight window, e.g. 20 -> 6
1037 return (hour >= openHour) || (hour < closeHour);
1038 } else {
1039 // openHour == closeHour: treat as always open
1040 return true;
1041 }
1042}

1044static int secondsUntilNextOpenForSeconds(uint32_t secondsOfDay,
1045 uint8_t openHour,
1046 uint8_t closeHour,
1047 bool openNow) {
1048 uint32_t openSec = (uint8_t)openHour * 3600;
1049 uint32_t closeSec = (uint8_t)closeHour * 3600;
1050
1051 // Normalize: if we're currently within opening hours, next open is tomorrow
1052 if (openNow) {
1053 return (int)((24 * 3600UL - secondsOfDay) + openSec);
1054 }
1055
1056 if (openHour < closeHour) {
1057 // Simple daytime window, closed before open or after close
1058 if (secondsOfDay < openSec) {
1059 // Before opening today
1060 return (int)(openSec - secondsOfDay);
1061 } else {
1062 // After closing, next open is tomorrow
1063 return (int)((24 * 3600UL - secondsOfDay) + openSec);
1064 }
1065 } else if (openHour > closeHour) {
1066 // Overnight window; closed between closeHour and openHour
1067 if (secondsOfDay < openSec && secondsOfDay >= closeSec) {
1068 // During the closed gap today
1069 return (int)(openSec - secondsOfDay);
1070 } else {
1071 // Otherwise next open is later today or tomorrow, but openNow
1072 // was already false so this path will generally be rare; fall back to 1 hour
1073 return 3600;
1074 }
1075 } else {
1076 // openHour == closeHour: always open; should not normally reach here
1077 return 3600;
1078 }
1079}

1083// Local time is derived from LocalTimeRK using the configured timezone.
1084// If time is not yet valid, we treat it as "open" so the device can start
1085// sensing while it acquires time and configuration.
1086bool isWithinOpenHours() {
1087 if (!Time.isValid()) {
1088 return true;

1094 const uint8_t hour = snapshot.localHour;
1095 const bool openNow = isWithinOpenHoursForHour(hour, openHour, closeHour);
1096
1097 return openNow;
1098}

1101int secondsUntilNextOpen() {
1102 if (!Time.isValid()) {

1161 // - Counting mode: keep the legacy "end of previous hour" timestamp so
1162 // hourly bucket aggregation behavior remains unchanged.

1177 float stateOfCharge = current.get_stateOfCharge();
1178 if (!(stateOfCharge == stateOfCharge) || stateOfCharge < 0.0f || stateOfCharge > 100.0f) {
1179 stateOfCharge = 0.0f;
1180 battState = 0;
1181 }

1184 if (sensorMode == OCCUPANCY) {
1185 const unsigned long timeStampValue = nowStampSec;

1188 // Occupancy mode webhook format (occupancy as 0/1 numeric value)
1189 snprintf(data, sizeof(data),
1190 "{\"occupancy\":%d,\"dailyoccupancy\":%lu,\"battery\":%4.2f,\"key1\":\"%s\",\"temp\":%4.2f,\"alerts\":%i,\"resets\":%i,\"connecttime\":%i,\"timestamp\":%lu000}",
1191 current.get_occupied() ? 1 : 0,

1195 current.get_internalTempC(),
1196 reportedAlertCode,
1197 sysStatus.get_resetCount(),
1198 sysStatus.get_lastConnectionDuration(),
1199 timeStampValue);
1200
1201 } else {
1202 const unsigned long timeStampValue = endOfPrevHourStampSec;
1203
1204 // Counting mode webhook format (original format)
1205 snprintf(data, sizeof(data),
1206 "{\"hourly\":%i,\"daily\":%i,\"battery\":%4.2f,\"key1\":\"%s\",\"temp\":%4.2f,\"resets\":%i,\"alerts\":%i,\"connecttime\":%i,\"timestamp\":%lu000}",
1207 current.get_hourlyCount(),
1208 current.get_dailyCount(),
1209 stateOfCharge,
1210 batteryContext[battState],
1211 current.get_internalTempC(),
1212 sysStatus.get_resetCount(),
1213 reportedAlertCode,
1214 sysStatus.get_lastConnectionDuration(),
1215 timeStampValue);
1216

1225 }

1228 // and that report is accepted by the publish queue, clear it locally.
1229 // If the underlying condition persists, it will be raised again.
1230 if (queued &&
1231 reportedAlertCode > 0 &&
1232 isAutoClearAfterReportAlert(reportedAlertCode) &&
1233 current.get_alertCode() == reportedAlertCode) {
1234 if (sysStatus.get_verboseMode()) {
1235 Log.info("Clearing alert %d after queueing report payload", (int)reportedAlertCode);
1236 }

1238 current.set_lastAlertTime(0);

1243 // Otherwise start it when CONNECTING_STATE reports a successful cloud connect.
1244 session.webhookExpectedOnConnect = true;
1245 if (Particle.connected()) {
1246 session.webhookExpectedOnConnect = false;
1247 session.awaitingWebhookResponse = true;
1248 session.webhookAwaitStartMs = millis();

1254 if (!ledgerOk) {
1255 // Data ledger publish failure; escalate via alert so the error
1256 // supervisor can decide on corrective action.
1257 current.raiseAlert(42);
1258 }
1259
1260 if (sensorMode == OCCUPANCY) {
1261 Log.info("Report: occ=%d totalMin=%lu alert=%d q=%d ledger=%s",
1262 current.get_occupied() ? 1 : 0,
1263 (unsigned long)(current.get_totalOccupiedSeconds() / 60UL),
1264 (int)reportedAlertCode,
1265 queued ? 1 : 0,
1266 ledgerState);
1267 } else {
1268 Log.info("Report: hourly=%d daily=%d alert=%d q=%d ledger=%s",
1269 (int)current.get_hourlyCount(),
1270 (int)current.get_dailyCount(),
1271 (int)reportedAlertCode,
1272 queued ? 1 : 0,
1273 ledgerState);
1274 }

1472#if CONNECTIVITY_FAILSAFE_TEST_MODE

1477 }
1478#endif
1479 return;
1480 }
1481

1483#if CONNECTIVITY_FAILSAFE_TEST_MODE

1486 failsafeDeferReasonName(FAILSAFE_DEFER_UPDATE_PENDING),
1487 stateShortName(state),
1488 System.updatesPending() ? 1 : 0);
1489 }
1490#endif
1491 return;
1492 }
1493
1494 if (!Time.isValid()) {
1495#if CONNECTIVITY_FAILSAFE_TEST_MODE
1496 if (claimFailsafeDeferLog(FAILSAFE_DEFER_INVALID_TIME)) {
1497 Log.info("FailsafeDefer: reason=%s",
1498 failsafeDeferReasonName(FAILSAFE_DEFER_INVALID_TIME));
1499 }
1500#endif
1501 return;
1502 }
1503
1504 const time_t lastConnection = sysStatus.get_lastConnection();
1505 if (lastConnection == 0) {
1506#if CONNECTIVITY_FAILSAFE_TEST_MODE
1507 if (claimFailsafeDeferLog(FAILSAFE_DEFER_NO_LAST_CONNECTION)) {
1508 Log.info("FailsafeDefer: reason=%s",
1509 failsafeDeferReasonName(FAILSAFE_DEFER_NO_LAST_CONNECTION));
1510 }
1511#endif
1512 return;
1513 }
1514
1515 const time_t now = Time.now();
1516 if (now <= lastConnection) {
1517 return;
1518 }
1519
1520 const time_t connectionAgeSec = now - lastConnection;
1521 if (connectionAgeSec < ConnectivityPolicy::CONNECTIVITY_FAILSAFE_STALE_SEC) {
1522 return;
1523 }
1524
1525 uint8_t currentStage = sysStatus.get_connectivityRecoveryStage();
1526 if (currentStage > 3) {
1527 currentStage = 0;
1528 }
1529 if (currentStage >= 3) {
1530 return;
1531 }
1532
1533 time_t lastAction = sysStatus.get_lastConnectivityRecoveryAction();
1534 if (lastAction > now) {
1535 lastAction = 0;
1536 }
1537
1538 const uint8_t nextStage = currentStage + 1;
1539 time_t requiredDelay = ConnectivityPolicy::CONNECTIVITY_FAILSAFE_COOLDOWN_SEC;
1540 if (nextStage >= 2) {
1541 requiredDelay += (time_t)connectivityFailsafeJitterSec(nextStage);
1542 }
1543
1544 if (currentStage != 0 && lastAction != 0 && (now - lastAction) < requiredDelay) {
1545 return;
1546 }
1547
1548 if (activeConnectAttemptWithinBudget()) {
1549 return;
1550 }
1551
1552 const BatteryTier tier = currentBatteryTierForFailsafe();
1553 const bool externalPowerPresent = connectivityFailsafeHasExternalPower();
1554 const bool lowBatteryHardActionBlocked =
1555 nextStage >= 2 &&
1556 !externalPowerPresent &&
1557 (sysStatus.get_lowBatteryMode() || tier == TIER_SURVIVAL);
1558
1559 if (lowBatteryHardActionBlocked) {
1560#if CONNECTIVITY_FAILSAFE_TEST_MODE
1561 if (claimFailsafeDeferLog(FAILSAFE_DEFER_LOW_BATTERY_HARD_STAGE_SUPPRESSED)) {
1562 Log.info("FailsafeDefer: reason=%s stage=%u tier=%s",
1563 failsafeDeferReasonName(FAILSAFE_DEFER_LOW_BATTERY_HARD_STAGE_SUPPRESSED),
1564 (unsigned)nextStage,
1565 batteryTierShortName(tier));
1566 }
1567#else
1568 Log.info("Failsafe: defer reason=low-battery stage=%u tier=%s",
1569 (unsigned)nextStage,
1570 batteryTierShortName(tier));
1571#endif
1572 persistConnectivityFailsafeState(currentStage, now, false);
1573 return;
1574 }
1575
1576 if (nextStage == 1) {
1577 current.raiseAlert(ConnectivityPolicy::CONNECTIVITY_FAILSAFE_ALERT);
1578 setAppBreadcrumb(BREADCRUMB_CONNECTIVITY_FAILSAFE);
1579 persistConnectivityFailsafeState(1, now, true);
1580 Log.info("Failsafe: stage=1 action=radio-reset age=%lds",
1581 (long)connectionAgeSec);
1582 Connectivity::requestFullDisconnectAndRadioOff();
1583 state = CONNECTING_STATE;
1584 return;
1585 }
1586
1587 if (nextStage == 2) {
1588 setAppBreadcrumb(BREADCRUMB_CONNECTIVITY_FAILSAFE_HARD);
1589 persistConnectivityFailsafeState(2, now, true);
1590 Log.info("Failsafe: stage=2 action=system-reset age=%lds",
1591 (long)connectionAgeSec);
1592 System.reset();
1593 return;
1594 }
1595
1596 setAppBreadcrumb(BREADCRUMB_CONNECTIVITY_FAILSAFE_HARD);
1597 persistConnectivityFailsafeState(3, now, true);
1598 Log.info("Failsafe: stage=3 action=deep-powerdown age=%lds",
1599 (long)connectionAgeSec);
1600 ab1805.deepPowerDown();
1601}
