71/*
72 * Architectural overview
73 * ----------------------
74 * - State machine: setup()/loop() implement a simple state machine
75 * (INITIALIZATION, CONNECTING, IDLE, SLEEPING, REPORTING, ERROR)
76 * that drives sensing, reporting, and power management.
77 * - Sensor abstraction: ISensor + SensorFactory + SensorManager allow
78 * different physical sensors (PIR, ultrasonic, etc.) behind one API.
79 * - Cloud configuration: the Cloud singleton uses Particle Ledger to
80 * merge product defaults (default-settings) with per-device overrides
81 * (device-settings), then applies the merged config to persistent data.
82 * - Data publishing: publishData() builds a JSON payload and sends it
83 * via PublishQueuePosix (webhook) and also updates the device-data
84 * ledger for Console visibility.
85 * - Connectivity: compile-time macros (Wiring_WiFi / Wiring_Cellular)
86 * select WiFi vs. cellular for radio control; Particle.connect() is
87 * used to bring up the cloud session on both.
88 */

194void setup() {
195 const int reason = System.resetReason();
196 const uint32_t reasonData = System.resetReasonData();
197 const uint8_t previousBreadcrumb = appBreadcrumb;
198 const uint32_t previousBreadcrumbMs = appBreadcrumbMs;
199 startupPreviousBreadcrumb = previousBreadcrumb;
200 startupPreviousBreadcrumbMs = previousBreadcrumbMs;
201 setAppBreadcrumb(BREADCRUMB_SETUP_START);
202 bootStormLastResetReason = (uint8_t)reason;
203 bootStormLastResetReasonData = reasonData;

205 const bool previousBootFailedEarly = bootInProgress;
206 // Mark this boot as in-progress immediately so any early reset on this pass
207 // is detectable on the next boot.
208 bootInProgress = true;
209
210 bool qualifiesForStormCount = false;
211 switch (reason) {
212 case RESET_REASON_NONE:
213#ifdef RESET_REASON_PANIC
214 case RESET_REASON_PANIC:
215#endif
216 case RESET_REASON_WATCHDOG:
217#ifdef RESET_REASON_UNKNOWN
218 case RESET_REASON_UNKNOWN:
219#endif
220#ifdef RESET_REASON_USER_APPLICATION
221 case RESET_REASON_USER_APPLICATION:
222#endif
223 qualifiesForStormCount = true;
224 break;
225 default:
226 break;
227 }
228
229 if (previousBootFailedEarly && qualifiesForStormCount) {
230 Log.warn("Previous boot reset early before setup completed (reason=%d data=%lu)",
231 reason,
232 (unsigned long)reasonData);
233 }
234
235 if (previousBootFailedEarly && qualifiesForStormCount) {
236 if (bootStormCount < 255) {
237 bootStormCount++;
238 }
239 }
240
241 if (Time.isValid()) {
242 const time_t now = Time.now();
243 if (bootStormWindowStart == 0) {
244 bootStormWindowStart = now;
245 } else if ((now - bootStormWindowStart) > 600) {
246 bootStormCount = (previousBootFailedEarly && qualifiesForStormCount) ? 1 : 0;
247 bootStormWindowStart = now;
248 }
249 }
250
251 if (bootStormCount >= 6 && previousBootFailedEarly && qualifiesForStormCount) {
252 bootStormTripCount++;
253 bootStormAlertPending = true;
254 Log.error("BOOT STORM: %u early resets detected (reason=%d)", bootStormCount, reason);
255 Connectivity::requestFullDisconnectAndRadioOff();
256 Particle.process();
257 SystemSleepConfiguration bootStormSleep;
258 bootStormSleep.mode(SystemSleepMode::ULTRA_LOW_POWER).duration(600000UL);
259 System.sleep(bootStormSleep);
260 Log.warn("BOOT STORM holdoff sleep returned unexpectedly - continuing boot");
261 }

263 // Wait for serial connection only in explicit DEV builds
264#if ALLOW_BLOCKING_SERIAL_WAITS
265 waitFor(Serial.isConnected, 10000);
266 delay(1000);
267#endif
268
269 Log.info("===== Firmware Version %s =====", FIRMWARE_VERSION);
270 Log.info("===== Release Notes: %s =====", FIRMWARE_RELEASE_NOTES);
271 Log.info("Previous app breadcrumb: %s (%u) at %lu ms",
272 appBreadcrumbName(previousBreadcrumb),
273 (unsigned)previousBreadcrumb,
274 (unsigned long)previousBreadcrumbMs);

423 // Track how often the device has been resetting so the error supervisor
424 // can apply backoffs and avoid permanent reset loops. Only count resets
425 // that are likely to be recoverable by firmware (pin/user/watchdog).
426 switch (reason) {
427 case RESET_REASON_PIN_RESET:
428 case RESET_REASON_USER:
429 case RESET_REASON_WATCHDOG:
430#ifdef RESET_REASON_USER_APPLICATION
431 case RESET_REASON_USER_APPLICATION:
432#endif
433 sysStatus.set_resetCount(sysStatus.get_resetCount() + 1);
434 break;
435 case RESET_REASON_UPDATE:
436 // After OTA firmware update, force connection to reload
437 // configuration from ledger. This ensures device-settings
438 // (operatingMode, etc.) override any stale FRAM values.

441 break;

443 // Waking from sleep. Alert 40 suppression for overnight hibernate

512 // This must be configured before we can make any open/close hour decisions.

529 Log.info("Timezone: %s, Local time: %s", tz, conv.format(TIME_FORMAT_DEFAULT).c_str());
530 Log.info("Open hours %02u:00-%02u:00, currently: %s",
531 sysStatus.get_openTime(), sysStatus.get_closeTime(),

536 if (System.resetReason() == RESET_REASON_POWER_MANAGEMENT) {
537 uint8_t localHour = (uint8_t)(conv.getLocalTimeHMS().toSeconds() / 3600);

556 // Initialize the sensor based on configuration using *local* time. This

558 Log.info("Initial connectionMode: %d (%s)", sysStatus.get_connectionMode(),

560 sysStatus.get_connectionMode() == 1 ? "INTERMITTENT" :
561 sysStatus.get_connectionMode() == 2 ? "DISCONNECTED" : "INTERMITTENT_KEEP_ALIVE");

563 if (!SensorManager::instance().isSensorReady()) {
564 if (isWithinOpenHours()) {
565 Log.info("Initializing sensor after timezone setup");
566 SensorManager::instance().initializeFromConfig();
567
568 if (!SensorManager::instance().isSensorReady()) {
569 Log.error("Sensor failed to initialize after timezone setup; connecting to report error");
570 state = CONNECTING_STATE;
571 }
572 } else {
573 Log.info("Outside opening hours at startup; sensor will remain powered down");
574 // Ensure carrier sensor power rails are actually turned off even if
575 // we skipped sensor initialization while closed.
576 Log.info("Startup CLOSED: forcing sensor power down before sleep");
577 SensorManager::instance().onEnterSleep();
578 Log.info("Sensor ready after startup power-down: %s", SensorManager::instance().isSensorReady() ? "true" : "false");
579 }

590 // setup reached stable completion; clear early-boot in-progress marker.
591 bootInProgress = false;
592 setAppBreadcrumb(BREADCRUMB_SETUP_COMPLETE);
593 Log.info("Startup complete");

597void loop() {
598 // If we remain alive for a stability window, clear boot-storm counters.
599 if (bootStormCount > 0 && millis() > 300000UL) {
600 bootStormCount = 0;
601 bootStormWindowStart = 0;
602 }
603
604 // Main state machine driving sensing, reporting, power management
605 switch (state) {
606 case IDLE_STATE:
607 handleIdleState();
608 break;
609
610 case SLEEPING_STATE:
611 handleSleepingState();
612 break;
613
614 case REPORTING_STATE:
615 handleReportingState();
616 break;
617
618 case CONNECTING_STATE:
619 handleConnectingState();
620 break;
621
622 case FIRMWARE_UPDATE_STATE:
623 handleFirmwareUpdateState();
624 break;
625
626 case ERROR_STATE:
627 handleErrorState();
628 break;
629 }
630
631 thrashGuard.loop(state, millis());
632
633 ab1805.loop(); // Keeps the RTC synchronized with the device clock
634
635 // Housekeeping for each transit of the main loop
636 current.loop();
637 sysStatus.loop();
638 sensorConfig.loop();
639
640 // Service deferred cloud work (ledger status publishes, etc.)

740static bool isWithinOpenHoursForHour(uint8_t hour, uint8_t openHour, uint8_t closeHour) {
741 if (openHour < closeHour) {
742 // Simple daytime window, e.g. 6 -> 22
743 return (hour >= openHour) && (hour < closeHour);
744 } else if (openHour > closeHour) {
745 // Overnight window, e.g. 20 -> 6
746 return (hour >= openHour) || (hour < closeHour);
747 } else {
748 // openHour == closeHour: treat as always open
749 return true;
750 }
751}

753static int secondsUntilNextOpenForSeconds(uint32_t secondsOfDay,
754 uint8_t openHour,
755 uint8_t closeHour,
756 bool openNow) {
757 uint32_t openSec = (uint8_t)openHour * 3600;
758 uint32_t closeSec = (uint8_t)closeHour * 3600;
759
760 // Normalize: if we're currently within opening hours, next open is tomorrow
761 if (openNow) {
762 return (int)((24 * 3600UL - secondsOfDay) + openSec);
763 }
764
765 if (openHour < closeHour) {
766 // Simple daytime window, closed before open or after close
767 if (secondsOfDay < openSec) {
768 // Before opening today
769 return (int)(openSec - secondsOfDay);
770 } else {
771 // After closing, next open is tomorrow
772 return (int)((24 * 3600UL - secondsOfDay) + openSec);
773 }
774 } else if (openHour > closeHour) {
775 // Overnight window; closed between closeHour and openHour
776 if (secondsOfDay < openSec && secondsOfDay >= closeSec) {
777 // During the closed gap today
778 return (int)(openSec - secondsOfDay);
779 } else {
780 // Otherwise next open is later today or tomorrow, but openNow
781 // was already false so this path will generally be rare; fall back to 1 hour
782 return 3600;
783 }
784 } else {
785 // openHour == closeHour: always open; should not normally reach here
786 return 3600;
787 }
788}

791// Helper to determine whether current *local* time is within park open hours.
792// Local time is derived from LocalTimeRK using the configured timezone.
793// If time is not yet valid, we treat it as "open" so the device can start
794// sensing while it acquires time and configuration.
795bool isWithinOpenHours() {
796 if (!Time.isValid()) {
797 return true;

810int secondsUntilNextOpen() {
811 if (!Time.isValid()) {
