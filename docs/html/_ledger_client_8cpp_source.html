92bool Cloud::areLedgersSynced() const {
93 // A ledger is considered synced once its lastSynced() timestamp becomes non-zero.
94 time_t defaultSync = defaultSettingsLedger.lastSynced();
95 time_t deviceSync = deviceSettingsLedger.lastSynced();
96 bool defaultSynced = (defaultSync > 0);
97 bool deviceSynced = (deviceSync > 0);
98
99 // Track the current connection's sync window explicitly. The helper must not
100 // reuse timing state from a previous sleep/connect cycle.
101 static unsigned long firstConnectedTime = 0;
102 static bool wasDisconnected = true;
103 static time_t lastObservedConnectionEpoch = 0;
104 static bool timeoutOutcomeLogged = false;
105 static bool readyOutcomeLogged = false;
106 unsigned long nowMs = millis();
107
108 if (Particle.connected()) {
109 time_t currentConnectionEpoch = sysStatus.get_lastConnection();
110 bool newConnectionObserved = wasDisconnected ||
111 (currentConnectionEpoch != 0 && currentConnectionEpoch != lastObservedConnectionEpoch);
112
113 if (newConnectionObserved) {
114 firstConnectedTime = nowMs;
115 wasDisconnected = false;
116 lastObservedConnectionEpoch = currentConnectionEpoch;
117 timeoutOutcomeLogged = false;
118 readyOutcomeLogged = false;
119 if (sysStatus.get_verboseMode()) {
120 Log.info("Connected - starting %lu ms ledger sync window",
121 ConnectivityPolicy::LEDGER_SYNC_TIMEOUT_MS);
122 }
123#if ENABLE_LEDGER_TRACE
124 Log.info("LedgerInputWindow: start defaultSynced=%d deviceSynced=%d timeout=%lu",
125 defaultSynced ? 1 : 0,
126 deviceSynced ? 1 : 0,
127 (unsigned long)ConnectivityPolicy::LEDGER_SYNC_TIMEOUT_MS);
128#endif
129 }
130
131 // If both ledgers are already synced for this connection, do not force the
132 // caller to wait out the remaining window. This is the key Alert 44 fix.
133 if (defaultSynced && deviceSynced) {
134 if (!readyOutcomeLogged) {
135 unsigned long elapsedSinceConnect = nowMs - firstConnectedTime;
136#if ENABLE_LEDGER_TRACE
137 Log.info("LedgerInputReady: elapsed=%lu default=%lu device=%lu",
138 elapsedSinceConnect,
139 (unsigned long)defaultSync,
140 (unsigned long)deviceSync);
141#else
142 (void)elapsedSinceConnect;
143#endif
144 readyOutcomeLogged = true;
145 }
146 return true;
147 }
148
149 // Give ledgers time to sync after connection (platform-specific timeout)
150 unsigned long elapsedSinceConnect = nowMs - firstConnectedTime;
151 if (elapsedSinceConnect > ConnectivityPolicy::LEDGER_SYNC_TIMEOUT_MS) {
152 LedgerData defaultData = defaultSettingsLedger.get();
153 LedgerData deviceData = deviceSettingsLedger.get();
154 bool defaultHasConfig = ledgerHasConfigContent(defaultData);
155 bool deviceHasConfig = ledgerHasConfigContent(deviceData);
156
157 if (defaultSynced && !deviceSynced && !deviceHasConfig) {
158 if (!timeoutOutcomeLogged && sysStatus.get_verboseMode()) {
159 Log.info("default-settings synced and device-settings is empty after %lu ms - assuming no device overrides",
160 elapsedSinceConnect);
161 }
162 if (!timeoutOutcomeLogged) {
163#if ENABLE_LEDGER_TRACE
164 Log.info("LedgerInputReady: elapsed=%lu default=%lu device=%lu empty-device=1",
165 elapsedSinceConnect,
166 (unsigned long)defaultSync,
167 (unsigned long)deviceSync);
168#endif
169 }
170 timeoutOutcomeLogged = true;
171 return true;
172 }
173
174 if (!defaultSynced && deviceSynced && !defaultHasConfig) {
175 if (!timeoutOutcomeLogged && sysStatus.get_verboseMode()) {
176 Log.info("device-settings synced and default-settings is empty after %lu ms - assuming no product defaults",
177 elapsedSinceConnect);
178 }
179 if (!timeoutOutcomeLogged) {
180#if ENABLE_LEDGER_TRACE
181 Log.info("LedgerInputReady: elapsed=%lu default=%lu device=%lu empty-default=1",
182 elapsedSinceConnect,
183 (unsigned long)defaultSync,
184 (unsigned long)deviceSync);
185#endif
186 }
187 timeoutOutcomeLogged = true;
188 return true;
189 }
190
191 // If either ledger has synced, both must sync
192 if (defaultSynced || deviceSynced) {
193 bool bothSynced = (defaultSynced && deviceSynced);
194 if (!bothSynced) {
195 if (!timeoutOutcomeLogged) {
196 Log.warn("Partial ledger sync after %lu ms: default=%lu device=%lu",
197 elapsedSinceConnect,
198 (unsigned long)defaultSync, (unsigned long)deviceSync);
199 Log.info("LedgerInputBlocked: elapsed=%lu defaultSynced=%d deviceSynced=%d",
200 elapsedSinceConnect,
201 defaultSynced ? 1 : 0,
202 deviceSynced ? 1 : 0);
203 timeoutOutcomeLogged = true;
204 }
205 }
206 return bothSynced;
207 }
208 // If neither has synced after timeout, assume they're empty and that's okay
209 if (!timeoutOutcomeLogged && sysStatus.get_verboseMode()) {
210 Log.info("No ledger data after %lu ms - assuming empty ledgers (OK)", elapsedSinceConnect);
211 }
212 if (!timeoutOutcomeLogged) {
213#if ENABLE_LEDGER_TRACE
214 Log.info("LedgerInputReady: elapsed=%lu default=%lu device=%lu empty=1",
215 elapsedSinceConnect,
216 (unsigned long)defaultSync,
217 (unsigned long)deviceSync);
218#endif
219 }
220 timeoutOutcomeLogged = true;
221 return true;
222 }
223 return false;
224 } else {
225 // Disconnected - reset for next connection
226 wasDisconnected = true;
227 lastObservedConnectionEpoch = 0;
228 timeoutOutcomeLogged = false;
229 return false;
230 }
