196inline OccupancyCloseResult closeOccupancySessionSafely(const char *path) {
197 OccupancyCloseResult result;
198 const bool occupied = current.get_occupied();
199 const int8_t alertCode = current.get_alertCode();
200 const bool timeValid = Time.isValid();
201 const time_t now = Time.now();
202 const time_t start = current.get_occupancyStartTime();
203 const uint32_t previousTotal = current.get_totalOccupiedSeconds();
204 result.totalSeconds = previousTotal;
205
206 bool durationComputable = false;
207 long long rawSessionSeconds = 0;
208 uint32_t sessionSeconds = 0;
209 bool invalid = !timeValid || start == 0;
210
211 if (timeValid && start != 0) {
212 time_t effectiveStart = start;
213 if (start > now && start <= now + 5) {
214 effectiveStart = now;
215 }
216 rawSessionSeconds = (long long)now - (long long)effectiveStart;
217 durationComputable = true;
218 }
219
220 if (timeValid && start > now + 5) {
221 invalid = true;
222 }
223
224 if (!invalid && durationComputable) {
225 sessionSeconds = (uint32_t)rawSessionSeconds;
226 const uint32_t newTotal = previousTotal + sessionSeconds;
227 if (sessionSeconds > 86400UL || newTotal > 86400UL) {
228 invalid = true;
229 } else {
230 current.set_totalOccupiedSeconds(newTotal);
231 result.valid = true;
232 result.sessionSeconds = sessionSeconds;
233 result.totalSeconds = newTotal;
234 }
235 } else if (!invalid) {
236 invalid = true;
237 }
238
239 if (invalid) {
240 char sessionBuf[24];
241 if (durationComputable) {
242 snprintf(sessionBuf, sizeof(sessionBuf), "%lld", rawSessionSeconds);
243 } else {
244 strncpy(sessionBuf, "na", sizeof(sessionBuf));
245 sessionBuf[sizeof(sessionBuf) - 1] = '\0';
246 }
247 Log.warn("OccAnom: path=%s now=%lu start=%lu prev=%lu dur=%s occ=%d alert=%d",
248 path ? path : "?",
249 (unsigned long)now,
250 (unsigned long)start,
251 (unsigned long)previousTotal,
252 sessionBuf,
253 occupied ? 1 : 0,
254 (int)alertCode);
255 }
256
257 current.set_occupied(false);
258 current.set_occupancyStartTime(0);
259 return result;
260}
