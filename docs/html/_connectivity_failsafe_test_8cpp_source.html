113int plannedClosedHoursSleepSec() {
114 if (!Time.isValid() || !Config::isValid(false) || isWithinOpenHours()) {
115 return -1;

130FailsafeDeferReason currentFailsafeEligibilityReason() {
131 if (sysStatus.get_connectionMode() == DISCONNECTED) {
132 return FAILSAFE_DEFER_DISCONNECTED_MODE;
133 }
134
135 if (state == FIRMWARE_UPDATE_STATE || System.updatesPending()) {
136 return FAILSAFE_DEFER_UPDATE_PENDING;
137 }
138
139 if (!Time.isValid()) {
140 return FAILSAFE_DEFER_INVALID_TIME;
141 }
142
143 const time_t lastConnection = sysStatus.get_lastConnection();
144 if (lastConnection == 0) {
145 return FAILSAFE_DEFER_NO_LAST_CONNECTION;
146 }
147
148 const time_t now = Time.now();
149 if (now > lastConnection) {
150 const time_t connectionAgeSec = now - lastConnection;
151 uint8_t currentStage = sysStatus.get_connectivityRecoveryStage();
152 if (currentStage > 3) {
153 currentStage = 0;
154 }
155
156 if (connectionAgeSec >= ConnectivityPolicy::CONNECTIVITY_FAILSAFE_STALE_SEC && currentStage < 3) {
157 time_t lastAction = sysStatus.get_lastConnectivityRecoveryAction();
158 if (lastAction > now) {
159 lastAction = 0;
160 }
161
162 const uint8_t nextStage = currentStage + 1;
163 time_t requiredDelay = ConnectivityPolicy::CONNECTIVITY_FAILSAFE_COOLDOWN_SEC;
164 if (nextStage >= 2) {
165 requiredDelay += (time_t)ConnectivityPolicy::CONNECTIVITY_FAILSAFE_JITTER_MAX_SEC;
166 }
167
168 const bool cooldownComplete =
169 (currentStage == 0 || lastAction == 0 || (now - lastAction) >= requiredDelay);
170
171 if (cooldownComplete) {
172 const BatteryTier tier = currentBatteryTierForFailsafeLocal();
173 const bool externalPowerPresent = connectivityFailsafeHasExternalPowerLocal();
174 const bool lowBatteryHardActionBlocked =
175 nextStage >= 2 && !externalPowerPresent &&
176 (sysStatus.get_lowBatteryMode() || tier == TIER_SURVIVAL);
177
178 if (lowBatteryHardActionBlocked) {
179 return FAILSAFE_DEFER_LOW_BATTERY_HARD_STAGE_SUPPRESSED;
180 }
181 }
182 }
183 }
184
185 return FAILSAFE_DEFER_NONE;
186}

207void logBootDiagnostics() {
208 char lastActionField[24];
209 char connectionAgeField[24];
210 formatFailsafeAgeField(lastActionField, sizeof(lastActionField),
211 sysStatus.get_lastConnectivityRecoveryAction());
212 formatFailsafeConnectionAgeField(connectionAgeField, sizeof(connectionAgeField),
213 sysStatus.get_lastConnection());
214
215 const FailsafeDeferReason reason = currentFailsafeEligibilityReason();
216 if (reason == FAILSAFE_DEFER_NONE) {
217 Log.info("FailsafeBoot: test=1 stale=%lds cd=%lds jit=%lds stage=%u cnt=%u lastAct=%s connAge=%s eligible=1",
218 (long)ConnectivityPolicy::CONNECTIVITY_FAILSAFE_STALE_SEC,
219 (long)ConnectivityPolicy::CONNECTIVITY_FAILSAFE_COOLDOWN_SEC,
220 (long)ConnectivityPolicy::CONNECTIVITY_FAILSAFE_JITTER_MAX_SEC,
221 (unsigned)sysStatus.get_connectivityRecoveryStage(),
222 (unsigned)sysStatus.get_connectivityRecoveryCount(),
223 lastActionField,
224 connectionAgeField);
