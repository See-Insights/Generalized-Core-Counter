323bool SensorManager::loop() {
324 if (!_sensor || !_sensor->isReady()) {
325 return false;
326 }
327
328 // V3.23: Most sensors (like PIR) are interrupt-driven and should be
329 // serviced on every pass through the main loop.
330 // Future polling sensors would use sensor.setting2 or setting3 for polling interval.
331 if (_sensor->usesInterrupt()) {
332 bool event = _sensor->loop();
333 if (event && sysStatus.get_verboseMode()) {
334 Log.info("SensorManager: event reported by interrupt-driven sensor");
335 }
336 return event;
337 }
338
339 // Polling mode - for future sensors that don't use interrupts
340 // Polling interval would come from sensor.setting2 (in milliseconds)
341 unsigned long currentTime = millis();
342 uint32_t pollingInterval = sensorConfig.get_sensorSetting2(); // Polling interval in ms
343
344 if (pollingInterval == 0) {
345 // No polling interval set, check every loop
346 return _sensor->loop();
347 }
348
349 if (currentTime - _lastPollTime >= pollingInterval) {
350 _lastPollTime = currentTime;
351 return _sensor->loop();
352 }
353
354 return false;
355}

398float SensorManager::tmp36TemperatureC(int adcValue) {
399 // Analog inputs have values from 0-4095, or
400 // 12-bit precision. 0 = 0V, 4095 = 3.3V, 0.0008 volts (0.8 mV) per unit
401 // The temperature sensor docs use millivolts (mV), so use 3300 as the factor
402 // instead of 3.3.
403 float mV = ((float)adcValue) * 3300 / 4095;
404
405 // According to the TMP36 docs:
406 // Offset voltage 500 mV, scaling 10 mV/deg C, output voltage at 25C = 750 mV
407 // (77F) The offset voltage is subtracted from the actual voltage, allowing
408 // negative temperatures with positive voltages.
409
410 // Example value=969 mV=780.7 tempC=28.06884765625 tempF=82.52392578125
411
412 // With the TMP36, with the flat side facing you, the pins are:
413 // Vcc | Analog Out | Ground
414 // You must put a 0.1 uF capacitor between the analog output and ground or
415 // you'll get crazy inaccurate values!
416 return (mV - 500) / 10;
417}

419bool SensorManager::readTmp112TemperatureC(float &tempC) {
420 // TMP112A default 7-bit I2C address is 0x48.
421 // Allow override at compile time for unusual board strapping.
422#if defined(MUON_TMP112_I2C_ADDR)
423 const uint8_t addr = (uint8_t)MUON_TMP112_I2C_ADDR;
424#else
425 const uint8_t addr = 0x48;
426#endif
427
428 // Temperature register pointer is 0x00.
429 const uint8_t tempReg = 0x00;
430
431 // Guard against interference with other I2C users (AB1805, FRAM, etc.).
432 Wire.lock();
433
434 Wire.beginTransmission(addr);
435 Wire.write(tempReg);
436 int status = Wire.endTransmission(false);
437 if (status != 0) {
438 Wire.unlock();
439 return false;
440 }
441
442 const uint8_t toRead = 2;
443 (void)Wire.requestFrom((int)addr, (int)toRead);
444 if (Wire.available() < toRead) {
445 Wire.unlock();
446 return false;
447 }
448
449 uint8_t msb = (uint8_t)Wire.read();
450 uint8_t lsb = (uint8_t)Wire.read();
451 Wire.unlock();
452
453 // TMP112A temperature is a signed 12-bit value left-justified in 16 bits.
454 // Resolution is 0.0625 C per LSB.
455 int16_t raw = (int16_t)((((uint16_t)msb) << 8) | (uint16_t)lsb);
456 raw = (int16_t)(raw >> 4);
457 // Sign-extend the 12-bit value (bit 11 is the sign after shifting).
458 if (raw & 0x0800) {
459 raw = (int16_t)(raw | 0xF000);
460 }
461
462 tempC = ((float)raw) * 0.0625f;
463 return true;
464}

731 // PMIC Health Monitoring & Smart Remediation (BQ24195 PMIC)
732 // =========================================================================
733 // Supported platforms: Boron (Gen 3 cellular with BQ24195 PMIC)
734 // Excluded platforms: M-SoM/Muon (uses Particle Power Module with MAX17043, not BQ24195)
735 //
736 // Detects charging faults (1Hz amber LED = fault register set) and attempts
737 // automatic recovery with escalating remediation levels to prevent thrashing.
738 //
739 // Alert Codes (auto-reported via webhook):
740 // 20 = PMIC Thermal Shutdown (critical - charging stopped due to temp)
741 // 21 = PMIC Charge Timeout (critical - safety timer expired, stuck charging)
742 // 23 = PMIC Battery Fault (major - general charging issue)
743 //
744 // Log-Only Diagnostics (NOT alerted - transient/normal conditions):
745 // Input Fault: VBUS out of range (solar undervoltage common, backend detects sustained issues)
746 //
747 // Remediation Strategy:
748 // Level 0: Monitor only (log diagnostics, raise alert)
749 // Level 1: Soft reset (cycle charging off/on after 2+ consecutive faults)
750 // Level 2: Power cycle with watchdog (after 3+ consecutive faults)
751 // Cooldown: 1 hour minimum between remediation attempts
752 // Auto-Clear: Resets all counters when charging returns to healthy state
753 //
754 // This prevents the common "loss of charge until power cycle" issue by
755 // detecting PMIC faults early and automatically attempting recovery before
756 // requiring manual intervention.
757 // =========================================================================

914 remediationPhase = 0;

918 if (remediationInProgress) {

920 case 1: {
921 // Level 1: disableCharging -> wait 500ms -> enableCharging
922 if (remediationPhase == 0) {
923 pmic.disableCharging();
924 remediationPhaseStartMs = now;
925 remediationPhase = 1;
926 } else if (now - remediationPhaseStartMs >= 500UL) {

929 remediationInProgress = false;
930 remediationActiveLevel = 0;
931 remediationPhase = 0;
932 lastRemediationAttempt = now;
933 }
934 break;
935 }
936 case 2: {
937 // Level 2: disableCharging -> wait 1000ms -> set watchdog -> enableCharging
938 if (remediationPhase == 0) {
939 pmic.disableCharging();
940 remediationPhaseStartMs = now;
941 remediationPhase = 1;
942 } else if (now - remediationPhaseStartMs >= 1000UL) {
943 // Set watchdog to force reset if charging doesn't recover
944 pmic.setWatchdog(0b01); // 40 seconds
945 pmic.enableCharging();
946 Log.info("PMIC: Charging re-enabled with watchdog supervision");

948 remediationActiveLevel = 0;
949 remediationPhase = 0;
950 remediationLevel = 0; // Reset level after power cycle attempt
951 lastRemediationAttempt = now;
952 }
953 break;
954 }
955 default:
956 remediationInProgress = false;
957 remediationActiveLevel = 0;
958 remediationPhase = 0;
959 break;
960 }
961 } else {
962 // Not in progress; decide whether to start a remediation attempt.
963 if (now - lastRemediationAttempt > REMEDIATION_COOLDOWN) {
964 // Escalate remediation level based on consecutive faults
965 if (consecutiveFaults >= 3 && remediationLevel < 2) {
966 remediationLevel = 2; // Escalate to power cycle reset
967 } else if (consecutiveFaults >= 2 && remediationLevel < 1) {
968 remediationLevel = 1; // Escalate to disable/enable charging
969 }
970
971 switch (remediationLevel) {
972 case 1:
973 Log.warn("PMIC: Attempting soft remediation - cycle charging (level 1)");
974 remediationInProgress = true;
975 remediationActiveLevel = 1;
976 remediationPhase = 0;
977 break;
978
979 case 2:
980 Log.error("PMIC: Attempting aggressive remediation - power cycle reset (level 2)");
981 remediationInProgress = true;
982 remediationActiveLevel = 2;
983 remediationPhase = 0;
984 break;
985
986 default:
987 Log.info("PMIC: Fault detected but remediation level 0 - monitoring only");
988 break;
989 }
990 } else {
991 unsigned long remainingCooldown = (REMEDIATION_COOLDOWN - (now - lastRemediationAttempt)) / 60000;
992 Log.info("PMIC: Fault detected but in cooldown period (%lu min remaining)", remainingCooldown);
993 }
994 }
995 }
996 } else {
997 // No faults detected - clear counters if charging is healthy
998 immediateChargeFaultResetConsumed = false;
999 immediateChargeFaultResetFaultReg = 0;
1000 if (consecutiveFaults > 0) {
1001 Log.info("PMIC: Charging healthy - clearing fault counters");
1002 consecutiveFaults = 0;
1003 remediationLevel = 0;
1004
1005 // Clear any pending remediation sequencing now that faults are gone.
1006 remediationInProgress = false;
1007 remediationActiveLevel = 0;
1008 remediationPhase = 0;
1009
1010 // Clear PMIC-related alerts if they were active
1011 int8_t currentAlert = current.get_alertCode();
1012 if (currentAlert >= 20 && currentAlert <= 23) {
1013 Log.info("PMIC: Clearing battery/charging alert %d - charging resumed", currentAlert);
1014 current.set_alertCode(0);
1015 current.set_lastAlertTime(0);
1016 }
1017 }

1020 // Clear when PMIC fault cleared and not in stuck fast-charging state
1021 int8_t currentAlert = current.get_alertCode();
1022 if (currentAlert == 21 && chargeStatus != 2) {
1023 // Alert 21 active, no PMIC fault, and not in fast charging - charge recovered
1024 Log.info("PMIC: clearing alert 21 after charge recovery");
1025 current.set_alertCode(0);
1026 current.set_lastAlertTime(0);
1027 }
