291bool SensorManager::loop() {
292 if (!_sensor || !_sensor->isReady()) {
293 return false;
294 }
295
296 // V3.23: Most sensors (like PIR) are interrupt-driven and should be
297 // serviced on every pass through the main loop.
298 // Future polling sensors would use sensor.setting2 or setting3 for polling interval.
299 if (_sensor->usesInterrupt()) {
300 bool event = _sensor->loop();
301 if (event && sysStatus.get_verboseMode()) {
302 Log.info("SensorManager: event reported by interrupt-driven sensor");
303 }
304 return event;
305 }
306
307 // Polling mode - for future sensors that don't use interrupts
308 // Polling interval would come from sensor.setting2 (in milliseconds)
309 unsigned long currentTime = millis();
310 uint32_t pollingInterval = sensorConfig.get_sensorSetting2(); // Polling interval in ms
311
312 if (pollingInterval == 0) {
313 // No polling interval set, check every loop
314 return _sensor->loop();
315 }
316
317 if (currentTime - _lastPollTime >= pollingInterval) {
318 _lastPollTime = currentTime;
319 return _sensor->loop();
320 }
321
322 return false;

366float SensorManager::tmp36TemperatureC(int adcValue) {
367 // Analog inputs have values from 0-4095, or
368 // 12-bit precision. 0 = 0V, 4095 = 3.3V, 0.0008 volts (0.8 mV) per unit
369 // The temperature sensor docs use millivolts (mV), so use 3300 as the factor
370 // instead of 3.3.
371 float mV = ((float)adcValue) * 3300 / 4095;
372
373 // According to the TMP36 docs:
374 // Offset voltage 500 mV, scaling 10 mV/deg C, output voltage at 25C = 750 mV
375 // (77F) The offset voltage is subtracted from the actual voltage, allowing
376 // negative temperatures with positive voltages.
377
378 // Example value=969 mV=780.7 tempC=28.06884765625 tempF=82.52392578125
379
380 // With the TMP36, with the flat side facing you, the pins are:
381 // Vcc | Analog Out | Ground
382 // You must put a 0.1 uF capacitor between the analog output and ground or
383 // you'll get crazy inaccurate values!
384 return (mV - 500) / 10;
385}

387bool SensorManager::readTmp112TemperatureC(float &tempC) {
388 // TMP112A default 7-bit I2C address is 0x48.
389 // Allow override at compile time for unusual board strapping.
390#if defined(MUON_TMP112_I2C_ADDR)
391 const uint8_t addr = (uint8_t)MUON_TMP112_I2C_ADDR;
392#else
393 const uint8_t addr = 0x48;
394#endif
395
396 // Temperature register pointer is 0x00.
397 const uint8_t tempReg = 0x00;
398
399 // Guard against interference with other I2C users (AB1805, FRAM, etc.).
400 Wire.lock();
401
402 Wire.beginTransmission(addr);
403 Wire.write(tempReg);
404 int status = Wire.endTransmission(false);
405 if (status != 0) {
406 Wire.unlock();
407 return false;
408 }
409
410 const uint8_t toRead = 2;
411 (void)Wire.requestFrom((int)addr, (int)toRead);
412 if (Wire.available() < toRead) {
413 Wire.unlock();
414 return false;
415 }
416
417 uint8_t msb = (uint8_t)Wire.read();
418 uint8_t lsb = (uint8_t)Wire.read();
419 Wire.unlock();
420
421 // TMP112A temperature is a signed 12-bit value left-justified in 16 bits.
422 // Resolution is 0.0625 C per LSB.
423 int16_t raw = (int16_t)((((uint16_t)msb) << 8) | (uint16_t)lsb);
424 raw = (int16_t)(raw >> 4);
425 // Sign-extend the 12-bit value (bit 11 is the sign after shifting).
426 if (raw & 0x0800) {
427 raw = (int16_t)(raw | 0xF000);
428 }
429
430 tempC = ((float)raw) * 0.0625f;
431 return true;
432}

701 // automatic recovery with escalating remediation levels to prevent thrashing.
702 //
703 // Alert Codes (auto-reported via webhook):
704 // 20 = PMIC Thermal Shutdown (critical - charging stopped due to temp)
705 // 21 = PMIC Charge Timeout (critical - safety timer expired, stuck charging)
706 // 23 = PMIC Battery Fault (major - general charging issue)

708 // Log-Only Diagnostics (NOT alerted - transient/normal conditions):
709 // Input Fault: VBUS out of range (solar undervoltage common, backend detects sustained issues)

712 // Level 0: Monitor only (log diagnostics, raise alert)

908 pmic.setWatchdog(0b01); // 40 seconds
909 pmic.enableCharging();
910 Log.info("PMIC: Charging re-enabled with watchdog supervision");

912 remediationActiveLevel = 0;
913 remediationPhase = 0;
914 remediationLevel = 0; // Reset level after power cycle attempt
915 lastRemediationAttempt = now;
916 }
917 break;
918 }
919 default:
920 remediationInProgress = false;
921 remediationActiveLevel = 0;
922 remediationPhase = 0;
923 break;
924 }
925 } else {
926 // Not in progress; decide whether to start a remediation attempt.
927 if (now - lastRemediationAttempt > REMEDIATION_COOLDOWN) {
928 // Escalate remediation level based on consecutive faults
929 if (consecutiveFaults >= 3 && remediationLevel < 2) {

932 remediationLevel = 1; // Escalate to disable/enable charging
933 }

935 switch (remediationLevel) {
936 case 1:
937 Log.warn("PMIC: Attempting soft remediation - cycle charging (level 1)");
938 remediationInProgress = true;
939 remediationActiveLevel = 1;
940 remediationPhase = 0;
941 break;
942
943 case 2:
944 Log.error("PMIC: Attempting aggressive remediation - power cycle reset (level 2)");
945 remediationInProgress = true;
946 remediationActiveLevel = 2;
947 remediationPhase = 0;
948 break;
949
950 default:
951 Log.info("PMIC: Fault detected but remediation level 0 - monitoring only");
952 break;
953 }
954 } else {
955 unsigned long remainingCooldown = (REMEDIATION_COOLDOWN - (now - lastRemediationAttempt)) / 60000;
956 Log.info("PMIC: Fault detected but in cooldown period (%lu min remaining)", remainingCooldown);
957 }
958 }
959 }
960 } else {
961 // No faults detected - clear counters if charging is healthy
962 immediateChargeFaultResetConsumed = false;
963 immediateChargeFaultResetFaultReg = 0;
964 if (consecutiveFaults > 0) {
965 Log.info("PMIC: Charging healthy - clearing fault counters");
966 consecutiveFaults = 0;
967 remediationLevel = 0;
968
969 // Clear any pending remediation sequencing now that faults are gone.
970 remediationInProgress = false;
971 remediationActiveLevel = 0;
972 remediationPhase = 0;
973
974 // Clear PMIC-related alerts if they were active
975 int8_t currentAlert = current.get_alertCode();
976 if (currentAlert >= 20 && currentAlert <= 23) {
977 Log.info("PMIC: Clearing battery/charging alert %d - charging resumed", currentAlert);
978 current.set_alertCode(0);
979 current.set_lastAlertTime(0);
980 }
981 }

1296 current.set_internalTempC(tempC);
1297
1298#else
1299 // Measure enclosure temperature using the TMP36 on the carrier board
1300 // (connected to TMP36_SENSE_PIN, typically A4).
1301 // Non-blocking sampling: spread 8 samples across multiple batteryState()
1302 // calls to avoid blocking the main loop. Each call takes one sample (~5µs ADC
1303 // read) until all samples are collected, then computes the average.
1304 if (tmp112Present) {
1305 // TMP112A already provided a temperature this cycle; skip TMP36 sampling.
1306 // This avoids unnecessary ADC activity on boards where both might exist.
1307 isItSafeToCharge();
1308 return current.get_stateOfCharge() > 20.0f;
1309 }
1310 pinMode(TMP36_SENSE_PIN, INPUT);
1311
1312 const int TMP36_SAMPLES = 8;
1313 static int sampleIndex = 0; // Tracks how many samples taken this cycle
1314 static int tmpRawSum = 0; // Running sum of ADC readings
1315
1316 if (sampleIndex < TMP36_SAMPLES) {
1317 // Take one ADC sample per call, accumulate into sum
1318 int v = analogRead(TMP36_SENSE_PIN);
1319 tmpRawSum += v;
1320 sampleIndex++;
1321
1322 // Not done yet; use previous temperature value and return early.
1323 // Caller can call batteryState() again on next loop to continue sampling.
1324 return current.get_stateOfCharge() > 20.0f;
1325 }
1326
1327 // All samples collected; compute average and reset for next cycle
1328 int tmpRaw = tmpRawSum / TMP36_SAMPLES;
1329 sampleIndex = 0;
1330 tmpRawSum = 0;
1331
1332 // Consider extremely low readings as "sensor not present". With a TMP36,
1333 // even very cold temperatures should still be around 100mV (roughly 120
1334 // ADC counts on a 3.3V/12-bit ADC), so an average below ~50 counts is
1335 // effectively 0V at the pin.
1336 bool sensorOk = (tmpRaw > 50 && tmpRaw < 4000);
1337 float tempC = tmp36TemperatureC(tmpRaw);
1338
1339 // If the TMP36 reading is clearly out of a plausible enclosure range
1340 // (for example, -50C from a raw 0 reading), or the sensor appears to be
1341 // disconnected, fall back to a prior stored value or a conservative
1342 // default so that charging guard rails and telemetry still operate with
1343 // a realistic value.
1344 if (!sensorOk || tempC < -20.0f || tempC > 80.0f) {
1345 float prev = current.get_internalTempC();
1346 float fallback = 25.0f; // conservative room-temperature default
1347
1348 if (prev > -20.0f && prev < 80.0f) {
1349 fallback = prev;
1350 }
1351
1352 Log.warn("TMP36 reading invalid or out of range (tmp36=%4.2f C, raw=%d, sensorOk=%s) - falling back to %4.2f C",
1353 (double)tempC, tmpRaw, sensorOk ? "true" : "false", (double)fallback);
1354 tempC = fallback;
1355 }
1356
1357 current.set_internalTempC(tempC);
1358
1359 // Optional debug: log enclosure temperature when verbose logging is enabled
1360 if (sysStatus.get_verboseMode()) {
1361 Log.info("Enclosure temperature (effective): %4.2f C (raw=%d)", (double)tempC, tmpRaw);
1362 }
1363
1364#endif // PLATFORM_ID == 32 || PLATFORM_ID == 34
1365
1366 // Apply temperature-based charging guard rails (see reference implementation).
1367 // On cellular platforms this will enable/disable PMIC charging based on
1368 // current.get_internalTempC(); on others it is a no-op.
1369 isItSafeToCharge();
1370
1371 // Convenience: indicate whether battery is in a healthy range.
1372 return current.get_stateOfCharge() > 20.0f;
1373}

1375bool SensorManager::isItSafeToCharge() // Returns a true or false if the battery
1376 // is in a safe charging range based on
1377 // enclosure temperature
1378{
1379 float temp = current.get_internalTempC();
1380 // Apply simple hysteresis around the recommended LiPo charge range
1381 // to avoid rapid toggling near the temperature boundaries. When
1382 // charging is currently allowed, we disable if temp < 0C or > 45C.
1383 // When charging is currently disallowed, we only re-enable once
1384 // temp has returned to a tighter 2C-43C window.
1385 static bool lastSafe = true;
1386#if HAL_PLATFORM_CELLULAR
1387 const bool previousSafe = lastSafe;
1388#endif
1389
1390 bool safe;
1391 if (lastSafe) {
1392 safe = !(temp < 0.0f || temp > 45.0f);
1393 } else {
1394 safe = !(temp < 2.0f || temp > 43.0f);
1395 }
1396 lastSafe = safe;
1397
1398#if HAL_PLATFORM_CELLULAR
1399 // On Boron (cellular Gen 3), a BQ24195 PMIC is available so we
1400 // actually enable/disable charging based on the enclosure
1401 // temperature.
1402 const bool safeStateChanged = (safe != previousSafe);
1403 PMIC pmic(true);
1404
1405 if (!safe) {
1406 pmic.disableCharging();
1407 current.set_batteryState(1); // Reflect that we are "Not Charging"
1408
1409 Log.warn("Charging disabled due to enclosure temperature: %4.2f C", (double)temp);
1410 } else {
1411 pmic.enableCharging();
1412
1413 if (safeStateChanged || sysStatus.get_verboseMode()) {
1414 Log.info("Charging enabled; enclosure temperature: %4.2f C", (double)temp);
1415 }
1416 }
1417#else
1418 // On platforms without a PMIC API (such as Argon, Photon 2 / P2), we
1419 // do not control charging, but we still evaluate and log whether it
1420 // would be considered safe based on the same temperature range.
1421 if (!safe) {
1422 Log.warn("Charging would be disabled due to enclosure temperature: %4.2f C (no PMIC on this platform)", (double)temp);
1423 } else if (sysStatus.get_verboseMode()) {
1424 Log.info("Charging would be enabled; enclosure temperature: %4.2f C (no PMIC on this platform)", (double)temp);
1425 }
1426#endif
1427
1428 return safe;
1429}
