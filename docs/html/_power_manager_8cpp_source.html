25PowerInputProfile selectInputProfile(
26 const PowerPlatform::PowerSourceSnapshot &snapshot,
27 PowerInputProfile fallbackProfile,
28 PowerInputProfile lastAppliedProfile,
29 PowerProfileSelectionReason &reason) {
30 switch (snapshot.source) {
31 case kPowerSourceUsbHost:
32 reason = PowerProfileSelectionReason::UsbPowerSource;
33 return PowerInputProfile::UsbBench;
34 case kPowerSourceUsbAdapter:
35 reason = PowerProfileSelectionReason::UsbPowerSource;
36 return PowerInputProfile::UsbBench;
37 case kPowerSourceUsbOtg:
38 reason = PowerProfileSelectionReason::UsbPowerSource;
39 return PowerInputProfile::UsbBench;
40 case kPowerSourceVin:
41 reason = PowerProfileSelectionReason::VinPowerSource;
42 return PowerInputProfile::Solar35W;
43 case kPowerSourceBattery:
44 if (lastAppliedProfile != PowerInputProfile::NotApplicable) {
45 reason = PowerProfileSelectionReason::BatteryKeepLast;
46 return lastAppliedProfile;
47 }
48 reason = PowerProfileSelectionReason::BatteryFallback;
49 return fallbackProfile;
50 default:
51 reason = PowerProfileSelectionReason::UnknownSourceFallback;
52 return fallbackProfile;
53 }

100bool PowerManager::refreshInputProfile() {
101 report_.capabilities = PowerPlatform::detectCapabilities();
102 report_.action.chargingControlAvailable = report_.capabilities.hasChargingControl;
103
104 PowerReport nextReport = report_;
105 nextReport.valid = true;
106 nextReport.powerConfigurationResult = SYSTEM_ERROR_NONE;
107
108 if (!nextReport.capabilities.hasPmicPowerConfiguration) {
109 nextReport.activeInputProfile = PowerInputProfile::NotApplicable;
110 nextReport.inputProfileReason =
111 PowerProfileSelectionReason::UnsupportedPlatform;
112 nextReport.inputProfileStatus = PowerAvailability::NotAvailable;
113 nextReport.reading.powerSource = -1;
114 nextReport.reading.powerSourceStatus = PowerAvailability::NotAvailable;
115
116 if (shouldLogProfileDecision(report_, nextReport)) {
117 Log.info("Power: prof=%s src=%s cfg=%d",
118 compactProfileLabel(nextReport.activeInputProfile),
119 powerSourceLabel(nextReport.reading.powerSource),
120 nextReport.powerConfigurationResult);
121 }
122
123 report_ = nextReport;
124 return true;
125 }
126
127 const PowerPlatform::PowerSourceSnapshot sourceSnapshot =
128 PowerPlatform::readPowerSource();
129 nextReport.reading.powerSource = sourceSnapshot.source;
130 nextReport.reading.powerSourceStatus = sourceSnapshot.status;
131
132 const PowerInputProfile fallbackProfile = configuredFallbackProfile();
133 PowerProfileSelectionReason selectionReason =
134 PowerProfileSelectionReason::Unknown;
135 const PowerInputProfile selectedProfile =
136 selectInputProfile(sourceSnapshot,
137 fallbackProfile,
138 report_.activeInputProfile,
139 selectionReason);
140 nextReport.activeInputProfile = selectedProfile;
141 nextReport.inputProfileReason = selectionReason;
142 nextReport.inputProfileStatus =
143 (selectedProfile == PowerInputProfile::NotApplicable)
144 ? PowerAvailability::NotAvailable
145 : ((sourceSnapshot.status == PowerAvailability::Unknown ||
146 sourceSnapshot.status == PowerAvailability::NotAvailable ||
147 sourceSnapshot.status == PowerAvailability::Fallback)
148 ? PowerAvailability::Fallback
149 : PowerAvailability::Valid);
150
151 const bool shouldApplyProfile =
152 !report_.valid || report_.activeInputProfile != selectedProfile;
153 if (shouldApplyProfile) {
154 const PowerPlatform::PowerConfigurationApplyResult applyResult =
155 PowerPlatform::applyInputProfile(selectedProfile);
156 nextReport.powerConfigurationResult = applyResult.systemResult;
157 if (!applyResult.applied) {
158 nextReport.inputProfileStatus = PowerAvailability::Fallback;
159 }
160 }
161
162 if (shouldLogProfileDecision(report_, nextReport)) {
163 Log.info("Power: prof=%s src=%s cfg=%d",
164 compactProfileLabel(nextReport.activeInputProfile),
165 powerSourceLabel(nextReport.reading.powerSource),
166 nextReport.powerConfigurationResult);
167 }
168
169 report_ = nextReport;
170 return report_.powerConfigurationResult == SYSTEM_ERROR_NONE;
171}
